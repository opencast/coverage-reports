37  /**
38   * Query for the events list.
39   *
40   * The following filters can be used:
41   * <ul>
42   * <li>series</li>
43   * <li>presenters</li>
44   * <li>presenter's usernames</li>
45   * <li>contributors</li>
46   * <li>location</li>
47   * <li>agent</li>
48   * <li>language</li>
49   * <li>startDate</li>
50   * <li>status</li>
51   * </ul>
52   */

102   public EventListQuery() {
103     super();
104     this.availableFilters.add(createSeriesFilter(Optional.empty()));
105     this.availableFilters.add(createLocationFilter(Optional.empty()));
106     this.availableFilters.add(createAgentFilter(Optional.empty()));
107     this.availableFilters.add(createLanguageFilter(Optional.empty()));
108     this.availableFilters.add(createStartDateFilter(Optional.empty()));
109     this.availableFilters.add(createStatusFilter(Optional.empty()));
110     this.availableFilters.add(createCommentsFilter(Optional.empty()));
111     this.availableFilters.add(createIsPublishedFilter(Optional.empty()));
112     this.availableFilters.add(createReadAccessFilter(Optional.empty()));
113     this.availableFilters.add(createWriteAccessFilter(Optional.empty()));
114     this.availableFilters.add(createNeedsCuttingFilter(Optional.empty()));
115   }

117   /**
118    * Add a {@link ResourceListFilter} filter to the query with the given seriesId
119    *
120    * @param seriesId
121    *          the seriesId to filter with
122    */
123   public void withSeriesId(String seriesId) {
124     this.addFilter(createSeriesFilter(Optional.ofNullable(seriesId)));
125   }

326   /**
327    * Create a new {@link ResourceListFilter} based on the Series id
328    *
329    * @param seriesId
330    *          the series id to filter on wrapped in an {@link Optional} or {@link Optional#empty()}
331    * @return a new {@link ResourceListFilter} for the Series name based query
332    */
333   public static ResourceListFilter<String> createSeriesFilter(Optional<String> seriesId) {
334     return FiltersUtils.generateFilter(seriesId, FILTER_SERIES_NAME, FILTER_SERIES_LABEL, SourceType.SELECT,
335             Optional.of(SeriesListProvider.PROVIDER_PREFIX));
336   }
