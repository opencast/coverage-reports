62    /**
63     * Create a generic DublinCoreCatalog from an OcDublinCore.
64     * Fields are encoded according to the Opencast rules. This class provides functions for each DublinCore
65     * property that needs special encoding.
66     */
67    public static DublinCoreCatalog toCatalog(final OcDublinCore source) {
68      // completeness assured by unit test
69      final DublinCoreCatalog target = DublinCores.mkOpencastEpisode().getCatalog();
70      if (source.getAbstract().isPresent()) {
71        target.set(DublinCore.PROPERTY_ABSTRACT, source.getAbstract().get());
72      }
73      if (source.getAccessRights().isPresent()) {
74        target.set(DublinCore.PROPERTY_ACCESS_RIGHTS, source.getAccessRights().get());
75      }
76      if (source.getAccrualMethod().isPresent()) {
77        target.set(DublinCore.PROPERTY_ACCRUAL_METHOD, source.getAccrualMethod().get());
78      }
79      if (source.getAccrualPeriodicity().isPresent()) {
80        target.set(DublinCore.PROPERTY_ACCRUAL_PERIODICITY, source.getAccrualPeriodicity().get());
81      }
82      if (source.getAccrualPolicy().isPresent()) {
83        target.set(DublinCore.PROPERTY_ACCRUAL_POLICY, source.getAccrualPolicy().get());
84      }
85      if (source.getAlternative().isPresent()) {
86        target.set(DublinCore.PROPERTY_ALTERNATIVE, source.getAlternative().get());
87      }
88      if (source.getAudience().isPresent()) {
89        target.set(DublinCore.PROPERTY_AUDIENCE, source.getAudience().get());
90      }
91      if (source.getAvailable().isPresent()) {
92        target.set(DublinCore.PROPERTY_AVAILABLE, source.getAvailable().get());
93      }
94      if (source.getBibliographicCitation().isPresent()) {
95        target.set(DublinCore.PROPERTY_BIBLIOGRAPHIC_CITATION, source.getBibliographicCitation().get());
96      }
97      if (source.getConformsTo().isPresent()) {
98        target.set(DublinCore.PROPERTY_CONFORMS_TO, source.getConformsTo().get());
99      }
100     if (source.getContributor().isPresent()) {
101       target.set(DublinCore.PROPERTY_CONTRIBUTOR, source.getContributor().get());
102     }
103     if (source.getCoverage().isPresent()) {
104       target.set(DublinCore.PROPERTY_COVERAGE, source.getCoverage().get());
105     }
106     target.set(DublinCore.PROPERTY_CREATED, encodeCreated(source.getCreated()));
107     if (source.getCreator().isPresent()) {
108       target.set(DublinCore.PROPERTY_CREATOR, source.getCreator().get());
109     }
110     if (source.getDate().isPresent()) {
111       target.set(DublinCore.PROPERTY_DATE, encodeDate(source.getDate().get()));
112     }
113     if (source.getDateAccepted().isPresent()) {
114       target.set(DublinCore.PROPERTY_DATE_ACCEPTED, encodeDateAccepted(source.getDateAccepted().get()));
115     }
116     if (source.getDateCopyrighted().isPresent()) {
117       target.set(DublinCore.PROPERTY_DATE_COPYRIGHTED, encodeDateCopyrighted(source.getDateCopyrighted().get()));
118     }
119     if (source.getDateSubmitted().isPresent()) {
120       target.set(DublinCore.PROPERTY_DATE_SUBMITTED, encodeDateSubmitted(source.getDateSubmitted().get()));
121     }
122     if (source.getDescription().isPresent()) {
123       target.set(DublinCore.PROPERTY_DESCRIPTION, source.getDescription().get());
124     }
125     if (source.getEducationLevel().isPresent()) {
126       target.set(DublinCore.PROPERTY_EDUCATION_LEVEL, source.getEducationLevel().get());
127     }
128     if (source.getExtent().isPresent()) {
129       target.set(DublinCore.PROPERTY_EXTENT, encodeExtent(source.getExtent().get()));
130     }
131     if (source.getFormat().isPresent()) {
132       target.set(DublinCore.PROPERTY_FORMAT, source.getFormat().get());
133     }
134     if (source.getHasFormat().isPresent()) {
135       target.set(DublinCore.PROPERTY_HAS_FORMAT, source.getHasFormat().get());
136     }
137     if (source.getHasPart().isPresent()) {
138       target.set(DublinCore.PROPERTY_HAS_PART, source.getHasPart().get());
139     }
140     if (source.getHasVersion().isPresent()) {
141       target.set(DublinCore.PROPERTY_HAS_VERSION, source.getHasVersion().get());
142     }
143     if (source.getIdentifier().isPresent()) {
144       target.set(DublinCore.PROPERTY_IDENTIFIER, source.getIdentifier().get());
145     }
146     if (source.getInstructionalMethod().isPresent()) {
147       target.set(DublinCore.PROPERTY_INSTRUCTIONAL_METHOD, source.getInstructionalMethod().get());
148     }
149     if (source.getIsFormatOf().isPresent()) {
150       target.set(DublinCore.PROPERTY_IS_FORMAT_OF, source.getIsFormatOf().get());
151     }
152     if (source.getIsPartOf().isPresent()) {
153       target.set(DublinCore.PROPERTY_IS_PART_OF, source.getIsPartOf().get());
154     }
155     if (source.getIsReferencedBy().isPresent()) {
156       target.set(DublinCore.PROPERTY_IS_REFERENCED_BY, source.getIsReferencedBy().get());
157     }
158     if (source.getIsReplacedBy().isPresent()) {
159       target.set(DublinCore.PROPERTY_IS_REPLACED_BY, source.getIsReplacedBy().get());
160     }
161     if (source.getIsRequiredBy().isPresent()) {
162       target.set(DublinCore.PROPERTY_IS_REQUIRED_BY, source.getIsRequiredBy().get());
163     }
164     if (source.getIssued().isPresent()) {
165       target.set(DublinCore.PROPERTY_ISSUED, source.getIssued().get());
166     }
167     if (source.getIsVersionOf().isPresent()) {
168       target.set(DublinCore.PROPERTY_IS_VERSION_OF, source.getIsVersionOf().get());
169     }
170     if (source.getLanguage().isPresent()) {
171       target.set(DublinCore.PROPERTY_LANGUAGE, source.getLanguage().get());
172     }
173     if (source.getLicense().isPresent()) {
174       target.set(DublinCore.PROPERTY_LICENSE, source.getLicense().get());
175     }
176     if (source.getMediator().isPresent()) {
177       target.set(DublinCore.PROPERTY_MEDIATOR, source.getMediator().get());
178     }
179     if (source.getMedium().isPresent()) {
180       target.set(DublinCore.PROPERTY_MEDIUM, source.getMedium().get());
181     }
182     if (source.getModified().isPresent()) {
183       target.set(DublinCore.PROPERTY_MODIFIED, source.getModified().get());
184     }
185     if (source.getProvenance().isPresent()) {
186       target.set(DublinCore.PROPERTY_PROVENANCE, source.getProvenance().get());
187     }
188     if (source.getPublisher().isPresent()) {
189       target.set(DublinCore.PROPERTY_PUBLISHER, source.getPublisher().get());
190     }
191     if (source.getReferences().isPresent()) {
192       target.set(DublinCore.PROPERTY_REFERENCES, source.getReferences().get());
193     }
194     if (source.getRelation().isPresent()) {
195       target.set(DublinCore.PROPERTY_RELATION, source.getRelation().get());
196     }
197     if (source.getReplaces().isPresent()) {
198       target.set(DublinCore.PROPERTY_REPLACES, source.getReplaces().get());
199     }
200     if (source.getRequires().isPresent()) {
201       target.set(DublinCore.PROPERTY_REQUIRES, source.getRequires().get());
202     }
203     if (source.getRights().isPresent()) {
204       target.set(DublinCore.PROPERTY_RIGHTS, source.getRights().get());
205     }
206     if (source.getRightsHolder().isPresent()) {
207       target.set(DublinCore.PROPERTY_RIGHTS_HOLDER, source.getRightsHolder().get());
208     }
209     if (source.getSource().isPresent()) {
210       target.set(DublinCore.PROPERTY_SOURCE, source.getSource().get());
211     }
212     if (source.getSpatial().isPresent()) {
213       target.set(DublinCore.PROPERTY_SPATIAL, source.getSpatial().get());
214     }
215     if (source.getSubject().isPresent()) {
216       target.set(DublinCore.PROPERTY_SUBJECT, source.getSubject().get());
217     }
218     if (source.getTableOfContents().isPresent()) {
219       target.set(DublinCore.PROPERTY_TABLE_OF_CONTENTS, source.getTableOfContents().get());
220     }
221     if (source.getTemporal().isPresent()) {
222       target.set(DublinCore.PROPERTY_TEMPORAL, source.getTemporal().get());
223     }
224     target.set(DublinCore.PROPERTY_TITLE, source.getTitle());
225     if (source.getType().isPresent()) {
226       target.set(DublinCore.PROPERTY_TYPE, source.getType().get());
227     }
228     if (source.getValid().isPresent()) {
229       target.set(DublinCore.PROPERTY_VALID, source.getValid().get());
230     }
231     return target;
232   }
