package org.opencastproject.graphql.playlist.type.output;
import graphql.annotations.annotationTypes.GraphQLDescription;
import graphql.annotations.annotationTypes.GraphQLField;
import graphql.annotations.annotationTypes.GraphQLName;
@GraphQLName(GqlDeletePlaylistPayload.TYPE_NAME)
@GraphQLDescription("The payload returned after deleting a playlist, containing the ID of the deleted playlist.")
public class GqlDeletePlaylistPayload {
  public static final String TYPE_NAME = "DeletePlaylistPayload";
  public GqlDeletePlaylistPayload() {
  }
  public GqlDeletePlaylistPayload(@GraphQLName("id") String id) {
    this.id = id;
  }
  @GraphQLField
  @GraphQLDescription("A unique identifier of the deleted playlist.")
  private String id;
  public String getId() {
    return id;
  }
  public void setId(String id) {
    this.id = id;
  }
}
