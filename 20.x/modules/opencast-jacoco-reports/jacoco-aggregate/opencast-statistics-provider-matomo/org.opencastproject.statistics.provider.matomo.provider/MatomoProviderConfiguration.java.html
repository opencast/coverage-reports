package org.opencastproject.statistics.provider.matomo.provider;

import org.opencastproject.statistics.api.DataResolution;
import org.opencastproject.statistics.api.ProviderConfiguration;
import org.opencastproject.statistics.api.ResourceType;

import com.google.gson.Gson;

import java.util.Set;

public class MatomoProviderConfiguration extends ProviderConfiguration {
  private static final Gson gson = new Gson();

  public static class MatomoProviderSource {
    private String siteId;
    private String method;
    private String dimensionId;
    private String aggregation;
    private String aggregationVariable;
    private Set<DataResolution> resolutions;

    public MatomoProviderSource() {
      // needed for gson
    }

    public MatomoProviderSource(
        String siteId,
        String method,
        String dimensionId,
        String aggregation,
        String aggregationVariable,
        Set<DataResolution> resolutions
    ) {
      this.siteId = siteId;
      this.method = method;
      this.dimensionId = dimensionId;
      this.aggregation = aggregation;
      this.aggregationVariable = aggregationVariable;
      this.resolutions = resolutions;
    }

    public String getSiteId() {
      return siteId;
    }

    public String getMethod() {
      return method;
    }

    public String getDimensionId() {
      return dimensionId;
    }

    public String getAggregation() {
      return aggregation;
    }

    public String getAggregationVariable() {
      return aggregationVariable;
    }

    public Set<DataResolution> getResolutions() {
      return resolutions;
    }
  }

  private Set<MatomoProviderSource> sources;

  public MatomoProviderConfiguration() {
    // needed for gson
  }

  public MatomoProviderConfiguration(
      String id,
      String title,
      String description,
      ResourceType resourceType,
      String type,
      Set<MatomoProviderSource> sources
  ) {
    super(id, title, description, resourceType, type);
    this.sources = sources;
  }

  public Set<MatomoProviderSource> getSources() {
    return sources;
  }

  public static MatomoProviderConfiguration fromJson(String json) {
    return gson.fromJson(json, MatomoProviderConfiguration.class);
  }
}
