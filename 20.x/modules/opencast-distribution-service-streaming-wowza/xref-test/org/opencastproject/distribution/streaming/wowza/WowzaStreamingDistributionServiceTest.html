354   /**
355    * A streaming element's URI is below the streaming URL, so its retraction target used to be
356    * derived from the URI. The target must be the SMIL file the service derives from the
357    * channel, the media package and the flavor instead, so that a crafted URI cannot steer the
358    * deletion to another file.
359    */
360   @Test
361   public void testRetractStreamingElementTargetsSmilFile() throws Exception {
362     File distRoot = new File("target/wowza-retract-test/streams");
363     FileUtils.deleteQuietly(distRoot.getParentFile());
364     activateStreamingService(distRoot);
365 
366     // A streaming element whose URI references a SMIL name that does not belong to this
367     // channel and media package
368     MediaPackage mp = MediaPackageParser.getFromXml(
369             "<mediapackage xmlns=\"http://mediapackage.opencastproject.org\" id=\"mp-1\">"
370                     + "<media><track id=\"track-1\" type=\"presenter/source\">"
371                     + "<url>http://wowza.example/live/smil:other.smil/playlist.m3u8</url>"
372                     + "</track></media></mediapackage>");
373 
374     File orgDir = new File(distRoot, tenant1);
375     File smilFile = new File(orgDir, "channel1_mp-1_presenter.smil");
376     File referencedFile = new File(orgDir, "channel1/mp-1/track-1/file.mp4");
377     File foreignSmil = new File(orgDir, "other.smil");
378     FileUtils.forceMkdir(referencedFile.getParentFile());
379     Files.writeString(referencedFile.toPath(), "video");
380     Files.writeString(smilFile.toPath(), smilXml("channel1/mp-1/track-1/file.mp4"));
381     Files.writeString(foreignSmil.toPath(), "must not be deleted");
382 
383     streamingService.retractSync("channel1", mp, "track-1");
384 
385     assertTrue("The SMIL file must be deleted", !smilFile.exists());
386     assertTrue("The file referenced by the SMIL file must be deleted", !referencedFile.exists());
387     assertTrue("A SMIL file named by the element URI must not be deleted", foreignSmil.exists());
388   }

421   /**
422    * The media package identifier is joined into the distribution path. A programmatic media
423    * package carrying an unsafe identifier must fail with an IllegalStateException, not with a
424    * bare IllegalArgumentException.
425    */
426   @Test
427   public void testRetractWithUnsafeMediaPackageIdentifierThrowsIllegalStateException() throws Exception {
428     File distRoot = new File("target/wowza-unsafe-id-test/streams");
429     FileUtils.deleteQuietly(distRoot.getParentFile());
430     activateStreamingService(distRoot);
431 
432     MediaPackage mp = MediaPackageParser.getFromXml(
433             "<mediapackage xmlns=\"http://mediapackage.opencastproject.org\" id=\"mp-1\">"
434                     + "<media><track id=\"track-1\" type=\"presenter/source\">"
435                     + "<url>http://wfr.example/mediapackage/mp-1/track-1.mp4</url>"
436                     + "</track></media></mediapackage>");
437     // '..' collapses the media package level out of the channel subtree, so the containment
438     // check must reject it
439     mp.setIdentifier(new IdImpl() {
440       {
441         this.id = "..";
442       }
443     });
444 
445     try {
446       streamingService.retractSync("channel1", mp, "track-1");
447       fail("Expected an unsafe media package identifier to be rejected");
448     } catch (IllegalStateException e) {
449       // expected
450     }
451   }

453   /**
454    * The channel identifier is joined into the distribution path and determines the directory
455    * the containment check is anchored at, so a traversal channel identifier would move that
456    * directory along with the path. It must be rejected before any file is touched.
457    */
458   @Test
459   public void testRetractWithUnsafeChannelIdIsRejected() throws Exception {
460     File distRoot = new File("target/wowza-unsafe-channel-test/a/b/streams");
461     FileUtils.deleteQuietly(new File("target/wowza-unsafe-channel-test"));
462     activateStreamingService(distRoot);
463 
464     MediaPackage mp = MediaPackageParser.getFromXml(
465             "<mediapackage xmlns=\"http://mediapackage.opencastproject.org\" id=\"mp-1\">"
466                     + "<media><track id=\"track-1\" type=\"presenter/source\">"
467                     + "<url>http://wfr.example/mediapackage/mp-1/track-1/file.mp4</url>"
468                     + "</track></media></mediapackage>");
469 
470     // <distRoot>/<org>/../../../mp-1/track-1/file.mp4 is a file outside the distribution directory
471     File escapee = new File(distRoot.getParentFile().getParentFile(), "mp-1/track-1/file.mp4");
472     FileUtils.forceMkdir(escapee.getParentFile());
473     Files.writeString(escapee.toPath(), "must not be deleted");
474 
475     for (String channelId : new String[] { "../../..", "..", ".", "", "a/b", "a\\b" }) {
476       try {
477         streamingService.retractSync(channelId, mp, "track-1");
478         fail("Expected the channel identifier '" + channelId + "' to be rejected");
479       } catch (IllegalStateException e) {
480         // expected
481       }
482     }
483     assertTrue("A file outside the distribution directory must not be deleted", escapee.exists());
484   }

486   /**
487    * An element whose URI is below the streaming URL is a streaming manifest. Its distribution
488    * file is the SMIL file, which must not be replaced with the content of the element when
489    * such an element is distributed.
490    */
491   @Test
492   public void testDistributeStreamingElementKeepsSmilFile() throws Exception {
493     File distRoot = new File("target/wowza-distribute-manifest-test/streams");
494     FileUtils.deleteQuietly(distRoot.getParentFile());
495     activateStreamingService(distRoot);
496 
497     // The workspace mock resolves the last URI segment against the test resources
498     MediaPackage mp = MediaPackageParser.getFromXml(
499             "<mediapackage xmlns=\"http://mediapackage.opencastproject.org\" id=\"mp-1\">"
500                     + "<media><track id=\"track-1\" type=\"presenter/source\">"
501                     + "<mimetype>video/mp4</mimetype>"
502                     + "<url>http://wowza.example/live/smil:other.smil/mediapackage.xml</url>"
503                     + "</track></media></mediapackage>");
504 
505     File smilFile = new File(new File(distRoot, tenant1), "channel1_mp-1_presenter.smil");
506     FileUtils.forceMkdir(smilFile.getParentFile());
507     String smil = smilXml("channel1/mp-1/track-1/file.mp4");
508     Files.writeString(smilFile.toPath(), smil);
509 
510     assertTrue(streamingService.distributeSync("channel1", mp, "track-1").isEmpty());
511     assertEquals("The SMIL file must not be overwritten", smil, Files.readString(smilFile.toPath()));
512   }

524   private static String smilXml(String src) {
525     return "<smil>\n"
526             + "  <head/>\n"
527             + "  <body>\n"
528             + "    <switch>\n"
529             + "      <video src=\"" + src + "\" bitrate=\"1000\"/>\n"
530             + "    </switch>\n"
531             + "  </body>\n"
532             + "</smil>\n";
