80    @Override
81    public void put(StoragePath storagePath, Source source) throws AssetStoreException {
82      // Retrieving the file from the workspace has the advantage that in most cases the file already exists in the local
83      // working file repository. In the very few cases where the file is not in the working file repository,
84      // this strategy leads to a minor overhead because the file not only gets downloaded and stored in the file system
85      // but also a hard link needs to be created (or if that's not possible, a copy of the file.
86      final File origin = getUniqueFileFromWorkspace(source);
87      final File destination = createFile(storagePath, source);
88      try {
89        mkParent(destination);
90        link(origin, destination);
91      } catch (IOException e) {
92        logger.error("Error while linking/copying file {} to {}: {}", origin, destination, getMessage(e));
93        throw new AssetStoreException(e);
94      } finally {
95        if (origin != null) {
96          FileUtils.deleteQuietly(origin);
97        }
98      }
99    }
100 
101   private File getUniqueFileFromWorkspace(Source source) {
102     try {
103       return getWorkspace().get(source.getUri(), true);
104     } catch (NotFoundException e) {
105       logger.error("Source file '{}' does not exist", source.getUri());
106       throw new AssetStoreException(e);
107     } catch (IOException e) {
108       logger.error("Error while getting file '{}' from workspace: {}", source.getUri(), getMessage(e));
109       throw new AssetStoreException(e);
110     }
111   }

113   @Override
114   public boolean copy(final StoragePath from, final StoragePath to) throws AssetStoreException {
115     var file = findStoragePathFile(from);
116     if (file.isPresent()) {
117       var f = file.get();
118 
119       final File t = createFile(to, f);
120       mkParent(t);
121       logger.debug("Copying {} to {}", f.getAbsolutePath(), t.getAbsolutePath());
122       try {
123         link(f, t, true);
124       } catch (IOException e) {
125         logger.error("Error copying archive file {} to {}", f, t);
126         throw new AssetStoreException(e);
127       }
128       return true;
129     }
130     return false;
131   }

133   @Override
134   public Optional<InputStream> get(final StoragePath path) throws AssetStoreException {
135     var file = findStoragePathFile(path);
136     if (file.isPresent()) {
137       try {
138         return Optional.of(new FileInputStream(file.get()));
139       } catch (FileNotFoundException e) {
140         logger.error("Error getting archive file {}", file);
141         throw new AssetStoreException(e);
142       }
143     }
144     return Optional.empty();
145   }

147   @Override
148   public boolean contains(StoragePath path) throws AssetStoreException {
149     return findStoragePathFile(path).isPresent();
150   }

152   @Override
153   public boolean delete(DeletionSelector sel) throws AssetStoreException {
154     // Reject unsafe identifiers before they are used for anything, including the lookup of
155     // the root directory, which joins them into a path as well
156     ensureDeletionSelectorComponents(sel);
157     // Resolve the root directory once, before anything is deleted. Deleting the directory below may remove the very
158     // directory a second lookup would search for, so the result must not be resolved again afterwards.
159     final AssetStoreDirectory rootDirectory = getRootDirectory(sel.getOrganizationId(), sel.getMediaPackageId());
160     if (rootDirectory == null) {
161       // MediaPackage could not be found locally. This could mean
162       //   - all snapshots live in a remote asset store
163       //   - mount failed and files are temporary not available
164       //   - file was deleted out-of-band
165       //   - other fs problem
166       // In any case, we cannot continue and return "false" to indicate that the files could not be found.
167       return false;
168     }
169     final File dir = getDeletionSelectorDir(rootDirectory, sel);
170     // The deletion target must be exactly the media package directory (or one of its version
171     // sub-directories) below the store root — never a parent of it. The organization and
172     // media package identifiers are joined into the path, so an identifier that is the
173     // current or parent directory reference ('.' or '..') or the empty string would collapse
174     // the path into the parent directory; the check verifies the exact path shape instead of
175     // relying on lexical containment alone.
176     ensureMediaPackageShape(dir, rootDirectory, sel.getOrganizationId(), sel.getMediaPackageId(),
177             sel.getVersion());
178     try {
179       FileUtils.deleteDirectory(dir);
180       // also delete the media package directory if all versions have been deleted. Prefix directories are kept.
181       boolean mpDirDeleted = FileSupport.deleteHierarchyIfEmpty(
182               rootDirectory.deletionLimit(sel.getOrganizationId(), sel.getMediaPackageId()).toFile(),
183               dir.getParentFile());
184       if (mpDirDeleted) {
185         onDeleteMediaPackage(sel.getOrganizationId(), sel.getMediaPackageId());
186       }
187       return true;
188     } catch (IOException e) {
189       logger.error("Error deleting directory from archive {}", dir);
190       throw new AssetStoreException(e);
191     }
192   }

194   /**
195    * Ensures the given file lies within the given root directory (see
196    * {@link FileSupport#ensureWithinRoot(File, File)}).
197    *
198    * @param target
199    *          the file to check
200    * @param root
201    *          the root directory the file must be contained in
202    * @throws AssetStoreException
203    *         if the file is not within the root directory
204    */
205   private void ensureWithinRoot(File target, String root) throws AssetStoreException {
206     try {
207       FileSupport.ensureWithinRoot(target, new File(root));
208     } catch (IllegalArgumentException e) {
209       throw new AssetStoreException("Path is outside the asset store root " + new File(root).getAbsolutePath(), e);
210     }
211   }

213   /**
214    * Ensures the given directory is exactly the media package directory below the given store
215    * root, or — if a version is present — one of its version sub-directories, and nothing else.
216    * The organization and media package identifiers are joined into the path, so an identifier
217    * that is the current or parent directory reference ({@code .} or {@code ..}) or the empty
218    * string would collapse the path into a parent (or sibling) directory. Such components are
219    * already rejected by {@link #ensureDeletionSelectorComponents(DeletionSelector)}, before
220    * any lexical normalization can hide them; here, the resulting path must have exactly the
221    * shape {@code <root>/<organization>/[<prefix>/]<mediaPackage>[/version]}.
222    *
223    * @param dir
224    *          the directory to check
225    * @param rootDirectory
226    *          the store root the directory is below
227    * @param organizationId
228    *          the organization identifier
229    * @param mediaPackageId
230    *          the media package identifier
231    * @param version
232    *          the version, present if the directory is a version sub-directory
233    * @throws AssetStoreException
234    *         if the directory does not have the expected shape
235    */
236   private void ensureMediaPackageShape(File dir, AssetStoreDirectory rootDirectory, String organizationId,
237           String mediaPackageId, Optional<Version> version) throws AssetStoreException {
238     // The path of the root directory is absolute and normalized, and the identifiers are
239     // resolved against it as they are
240     Path expected = rootDirectory.mediaPackageDir(organizationId, mediaPackageId);
241     if (version.isPresent()) {
242       expected = expected.resolve(version.get().toString());
243     }
244     Path actual = dir.toPath().toAbsolutePath().normalize();
245     if (!actual.equals(expected)) {
246       throw new AssetStoreException("Deletion target " + actual + " does not point to the media package "
247               + mediaPackageId + " (expected " + expected + ")");
248     }
249   }

251   /**
252    * Whether the given string is unsafe to use as a single file system path component, i.e. it
253    * is the current or parent directory reference or the empty string, or it contains a path
254    * separator.
255    */
256   private static boolean isUnsafePathComponent(String component) {
257     return component == null || component.isEmpty() || ".".equals(component) || "..".equals(component)
258             || component.indexOf('/') >= 0 || component.indexOf('\\') >= 0;
259   }
260 
261   /**
262    * Rejects deletion selectors whose organization or media package identifiers are unsafe as
263    * file system path components (see {@link #isUnsafePathComponent(String)}).
264    */
265   private void ensureDeletionSelectorComponents(DeletionSelector sel) throws AssetStoreException {
266     if (isUnsafePathComponent(sel.getOrganizationId()) || isUnsafePathComponent(sel.getMediaPackageId())) {
267       throw new AssetStoreException("Unsafe organization or media package identifier in deletion path: '"
268               + sel.getOrganizationId() + "/" + sel.getMediaPackageId() + "'");
269     }
270   }
271 
272   /**
273    * Rejects storage paths whose organization, media package or element identifiers are unsafe
274    * as file system path components (see {@link #isUnsafePathComponent(String)}), so that the
275    * path below the store root cannot collapse into a parent (or sibling) directory.
276    */
277   private void ensureStoragePathComponents(StoragePath p) throws AssetStoreException {
278     if (isUnsafePathComponent(p.getOrganizationId()) || isUnsafePathComponent(p.getMediaPackageId())
279             || isUnsafePathComponent(p.getMediaPackageElementId())) {
280       throw new AssetStoreException(
281               "Unsafe organization, media package or element identifier in storage path: '"
282                       + p.getOrganizationId() + "/" + p.getMediaPackageId() + "/" + p.getMediaPackageElementId()
283                       + "'");
284     }
285   }

287   /**
288    * Returns the directory file from a deletion selector
289    *
290    * @param rootDirectory
291    *          the root directory of the media package the selector refers to
292    * @param sel
293    *          the deletion selector
294    * @return the directory file
295    */
296   private File getDeletionSelectorDir(AssetStoreDirectory rootDirectory, DeletionSelector sel) {
297     final Path mpDir = rootDirectory.mediaPackageDir(sel.getOrganizationId(), sel.getMediaPackageId());
298     if (sel.getVersion().isPresent()) {
299       return mpDir.resolve(sel.getVersion().get().toString()).toFile();
300     }
301     return mpDir.toFile();
302   }

304   /** Create all parent directories of a file. */
305   private void mkParent(File f) {
306     mkDirs(f.getParentFile());
307   }
308 
309   /** Create this directory and all of its parents. */
310   protected void mkDirs(File d) {
311     if (d == null) {
312       return;
313     }
314     // mkdirs *may* not have succeeded if it returns false, so check if the directory exists in that case
315     if (!d.mkdirs() && !d.exists()) {
316       final String msg = "Cannot create directory " + d;
317       logger.error(msg);
318       throw new AssetStoreException(msg);
319     }
320   }

322   /** Return the extension of a file. */
323   private Optional<String> extension(File f) {
324     Optional<String> opt = trimToNone(getExtension(f.getAbsolutePath()));
325     return opt.isPresent()
326         ? Optional.of(opt.get())
327         : Optional.empty();
328   }
329 
330   /** Return the extension of a URI, i.e. the extension of its path. */
331   private Optional<String> extension(URI uri) {
332     try {
333       Optional<String> opt = trimToNone(getExtension(uri.toURL().getPath()));
334       return opt.isPresent()
335           ? Optional.of(opt.get())
336           : Optional.empty();
337     } catch (MalformedURLException e) {
338       throw new Error(e);
339     }
340   }

352   /** Create a file from a storage path and an optional extension. */
353   private File createFile(StoragePath p, Optional<String> extension) {
354     ensureStoragePathComponents(p);
355     AssetStoreDirectory rootDirectory = getRootDirectory(p.getOrganizationId(), p.getMediaPackageId());
356     if (rootDirectory == null) {
357       rootDirectory = getRootDirectory();
358     }
359     final File destination = file(rootDirectory, p, extension);
360     ensureWithinRoot(destination, rootDirectory.getPath());
361     return destination;
362   }
363 
364   /** Returns a file from a storage path if its media package exists locally, null otherwise */
365   private File getExistingFile(StoragePath p, Optional<String> extension) {
366     ensureStoragePathComponents(p);
367     final AssetStoreDirectory rootDirectory = getRootDirectory(p.getOrganizationId(), p.getMediaPackageId());
368     if (rootDirectory == null) {
369       return null;
370     }
371     final File existingFile = file(rootDirectory, p, extension);
372     ensureWithinRoot(existingFile, rootDirectory.getPath());
373     return existingFile;
374   }
375 
376   /** The file of a storage path within a root directory */
377   private File file(AssetStoreDirectory rootDirectory, StoragePath p, Optional<String> extension) {
378     return rootDirectory.mediaPackageDir(p.getOrganizationId(), p.getMediaPackageId())
379             .resolve(p.getVersion().toString())
380             .resolve(extension.isPresent()
381                 ? p.getMediaPackageElementId() + EXTENSION_SEPARATOR + extension.get()
382                 : p.getMediaPackageElementId())
383             .toFile();
384   }

386   /**
387    * Returns a file {@link Optional} from a storage path if one is found or an empty {@link Optional}
388    *
389    * @param storagePath
390    *          the storage path
391    * @return the file {@link Optional}
392    */
393   private Optional<File> findStoragePathFile(final StoragePath storagePath) {
394     final FilenameFilter filter = new FilenameFilter() {
395       @Override
396       public boolean accept(File dir, String name) {
397         return FilenameUtils.getBaseName(name).equals(storagePath.getMediaPackageElementId());
398       }
399     };
400     final File elementFile = getExistingFile(storagePath, Optional.empty());
401     if (elementFile == null) {
402       // The media package does not exist locally, e.g. because all its snapshots live in a remote asset store
403       return Optional.empty();
404     }
405 
406     var files = elementFile.getParentFile().listFiles(filter);
407     if (files == null) {
408       return Optional.empty();
409     }
410     switch (files.length) {
411       case 0:
412         return Optional.empty();
413       case 1:
414         return Optional.of(files[0]);
415       default:
416         throw new AssetStoreException("Storage path " + files[0].getParent()
417             + "contains multiple files with the same element id!: " + storagePath.getMediaPackageElementId());
418     }
419   }
