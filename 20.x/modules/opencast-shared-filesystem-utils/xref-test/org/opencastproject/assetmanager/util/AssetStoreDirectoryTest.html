54    @Test
55    public void testPrefix() {
56      assertEquals("", AssetStoreDirectory.prefix("3f6e3c1a-8d0b-4b0e-9a64-0c2f0e8f7d21", 0));
57      assertEquals("3", AssetStoreDirectory.prefix("3f6e3c1a-8d0b-4b0e-9a64-0c2f0e8f7d21", 1));
58      assertEquals("3f", AssetStoreDirectory.prefix("3f6e3c1a-8d0b-4b0e-9a64-0c2f0e8f7d21", 2));
59      assertEquals("le", AssetStoreDirectory.prefix("Lecture-2026", 2));
60      assertEquals("__", AssetStoreDirectory.prefix("..abc", 2));
61      assertEquals("a_b", AssetStoreDirectory.prefix("a:b", 3));
62      assertEquals("x_", AssetStoreDirectory.prefix("x", 2));
63      assertEquals("___", AssetStoreDirectory.prefix("", 3));
64      // Non-ASCII characters are always replaced, even if Character.toLowerCase() would map them to ASCII
65      assertEquals("__a", AssetStoreDirectory.prefix("İKa", 3));
66      assertEquals("_", AssetStoreDirectory.prefix("ä", 1));
67    }

69    @Test
70    public void testPrefixLengthLimit() {
71      new AssetStoreDirectory("/mnt/archive", AssetStoreDirectory.MAX_PREFIX_LENGTH);
72      assertThrows(IllegalArgumentException.class,
73          () -> new AssetStoreDirectory("/mnt/archive", AssetStoreDirectory.MAX_PREFIX_LENGTH + 1));
74      assertThrows(IllegalArgumentException.class, () -> AssetStoreDirectory.fromConfig(bundleContext(Map.of(
75          ROOT_DIR, "/mnt/archive",
76          ROOT_DIR + PREFIX_LENGTH, "300"))));

116   @Test
117   public void testFromConfigMultipleDirectories() {
118     final List<AssetStoreDirectory> directories = AssetStoreDirectory.fromConfig(bundleContext(Map.of(
119         STORAGE_DIR, "/srv/opencast",
120         ROOT_DIR, "/mnt/archive ",
121         ROOT_DIR + PREFIX_LENGTH, "1",
122         ROOT_DIR + ".1", "/mnt/archive-1",
123         ROOT_DIR + ".2", "/mnt/archive-2",
124         ROOT_DIR + ".2" + PREFIX_LENGTH, " 2 ",
125         // not read because .3 is missing
126         ROOT_DIR + ".4", "/mnt/archive-4")));
127     assertEquals(3, directories.size());
128     assertEquals(new AssetStoreDirectory("/mnt/archive", 1), directories.get(0));
129     assertEquals(new AssetStoreDirectory("/mnt/archive-1", 0), directories.get(1));
130     assertEquals(new AssetStoreDirectory("/mnt/archive-2", 2), directories.get(2));
131   }

144   @Test
145   public void testMarkerIsWrittenForEmptyDirectory() throws Exception {
146     final File root = tmpFolder.newFolder();
147     final AssetStoreDirectory directory = new AssetStoreDirectory(root.getPath(), 2);
148     assertEquals(Optional.empty(), directory.readMarker());
149 
150     directory.verifyOrWriteMarker();
151     assertTrue(new File(root, AssetStoreDirectory.MARKER_FILE).isFile());
152     assertEquals(Optional.of(2), directory.readMarker());
153     // No temporary file is left behind
154     assertEquals(1, root.list().length);
155 
156     // Verifying again with the same configuration is fine
157     directory.verifyOrWriteMarker();
158   }

160   @Test
161   public void testMarkerIgnoresMountPointDirectories() throws Exception {
162     final File root = tmpFolder.newFolder();
163     assertTrue(new File(root, "lost+found").mkdir());
164     assertTrue(new File(root, ".snapshot").mkdir());
165     assertTrue(new File(root, "some-file").createNewFile());
166 
167     new AssetStoreDirectory(root.getPath(), 2).verifyOrWriteMarker();
168     assertEquals(Optional.of(2), new AssetStoreDirectory(root.getPath(), 2).readMarker());
169   }

180   @Test
181   public void testMarkerForLegacyData() throws Exception {
182     final File root = tmpFolder.newFolder();
183     assertTrue(new File(root, "mh_default_org/some-media-package/0").mkdirs());
184 
185     // Existing data without a marker file is in the flat layout
186     assertThrows(ConfigurationException.class, () -> new AssetStoreDirectory(root.getPath(), 2).verifyOrWriteMarker());
187     assertFalse(new File(root, AssetStoreDirectory.MARKER_FILE).exists());
188 
189     new AssetStoreDirectory(root.getPath(), 0).verifyOrWriteMarker();
190     assertEquals(Optional.of(0), new AssetStoreDirectory(root.getPath(), 0).readMarker());
191   }

193   @Test
194   public void testInvalidMarker() throws Exception {
195     final File root = tmpFolder.newFolder();
196     final AssetStoreDirectory directory = new AssetStoreDirectory(root.getPath(), 0);
197 
198     Files.writeString(new File(root, AssetStoreDirectory.MARKER_FILE).toPath(), "");
199     assertThrows(ConfigurationException.class, directory::readMarker);
200     assertThrows(ConfigurationException.class, directory::verifyOrWriteMarker);
201 
202     Files.writeString(new File(root, AssetStoreDirectory.MARKER_FILE).toPath(), "prefix.length=many");
203     assertThrows(ConfigurationException.class, directory::readMarker);
204   }

206   private static BundleContext bundleContext(Map<String, String> properties) {
207     final BundleContext bundleContext = EasyMock.createNiceMock(BundleContext.class);
208     EasyMock.expect(bundleContext.getProperty(EasyMock.anyString()))
209         .andAnswer(() -> properties.get((String) EasyMock.getCurrentArguments()[0])).anyTimes();
210     EasyMock.replay(bundleContext);
211     return bundleContext;
212   }
