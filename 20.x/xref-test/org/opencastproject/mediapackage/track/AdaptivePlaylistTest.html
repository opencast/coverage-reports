155   public Track makeTrack(URI uri, String logicalName) throws JAXBException, IOException, SAXException {
156     String xml = "<oc:track xmlns:oc=\"http://mediapackage.opencastproject.org\" "
157         + "type=\"presentation/source\"><oc:tags/><oc:url>"
158         + uri + "</oc:url><oc:duration>1</oc:duration><oc:logicalname>" + logicalName
159         + "</oc:logicalname></oc:track>";
160     InputStream inputStream = new ByteArrayInputStream(xml.getBytes(StandardCharsets.UTF_8));
161     TrackImpl track;
162     try {
163       track = unmarshaller.unmarshal(XmlSafeParser.parse(inputStream), TrackImpl.class).getValue();
164     } finally {
165       IoSupport.closeQuietly(inputStream);
166     }
167     return track;
168   }

170   protected HashMap<File, File> hlsmoveFiles(List<File> files) throws Exception, IOException {
171     // rename all the segments to match encoding profile
172     HashMap<File, File> outputs = new HashMap<File, File>();
173     for (File file : files) {
174       File destFile = File.createTempFile(FilenameUtils.getBaseName(file.getName()),
175               "." + FilenameUtils.getExtension(file.getName()),
176               file.getAbsoluteFile().getParentFile());
177       FileUtils.copyFile(file, destFile);
178       outputs.put(file, destFile);
179     }
180     return outputs;
181   }

227   @Test
228   public void testFileReferences() throws Exception {
229     Set<String> variants = AdaptivePlaylist.getVariants(master2);
230     assertEquals(variants.size(), 2);
231     Set<String> segments = AdaptivePlaylist.getVariants(variantFile);
232     assertEquals(segments.size(), 1);
233     Set<String> allsegments = AdaptivePlaylist.getReferencedFiles(master2, true);
234     assertEquals(allsegments.size(), 2);
235     Set<String> allFiles = AdaptivePlaylist.getReferencedFiles(master2, false);
236     assertEquals(allFiles.size(), 4);
237     List<File> files = new ArrayList<File>();
238     files.add(master2);
239     files.add(variantFile);
240     files.add(variantFile1);
241     files.add(srcFile);
242     files.add(srcFile1);
243     HashMap<File, File> movedFiles = hlsmoveFiles(files);
244     HashMap<String, String> nameMap = new HashMap<String, String>();
245     movedFiles.forEach((k, v) -> {
246       nameMap.put(k.getName(), v.getName());
247     });
248     for (File f : movedFiles.values()) {
249       if (AdaptivePlaylist.isPlaylist(f)) {
250         File destFile = File.createTempFile(f.getName(), FilenameUtils.getExtension(f.getName()),
251                 f.getAbsoluteFile().getParentFile());
252         Set<String> oldref = AdaptivePlaylist.getVariants(f);
253         AdaptivePlaylist.hlsRewriteFileReference(f, destFile, nameMap);
254         variants = AdaptivePlaylist.getVariants(destFile);
255         assertTrue(variants.size() > 0);
256         // the new name is used in the playlist
257         assertTrue(variants.contains(nameMap.get(oldref.toArray()[0])));
258       }
259     }
260     for (File f : movedFiles.values()) {
261       FileUtil.delete(f);
262     }
263   }
