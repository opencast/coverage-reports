63  /**
64   * This workflow operation allows to shift the timestamps of subtitle files.
65   * For example: If someone adds a bumper/intro video in front of an already subtitled presenter track
66   * the subtitles would start too early. With this operation, you can select a video and a subtitle track and the
67   * timestamps of the subtitle file will be shifted backwards by the duration of the selected video.
68   */
69  @Component(
70      property = {
71          "service.description=subtitle-timeshift Workflow Operation Handler",
72          "workflow.operation=subtitle-timeshift"
73      },
74      immediate = true,
75      service = WorkflowOperationHandler.class
76  )
77  public class SubtitleTimeshiftWorkflowOperationHandler extends AbstractWorkflowOperationHandler {
78  
79    private static final String SUBTITLE_SOURCE_FLAVOR_CFG_KEY = "subtitle-source-flavor";
80    private static final String VIDEO_SOURCE_FLAVOR_CFG_KEY = "video-source-flavors";
81    private static final String TARGET_FLAVOR_CFG_KEY = "target-flavor";
82  
83    /** The workspace collection name */
84    private static final String COLLECTION = "subtitles";

86    /** The logging facility */
87    private static final Logger logger = LoggerFactory.getLogger(SubtitleTimeshiftWorkflowOperationHandler.class);

93     */
94    @Reference
95    public void setWorkspace(Workspace workspace) {
96      this.workspace = workspace;
97    }

100   @Override
101   public WorkflowOperationResult start(WorkflowInstance workflowInstance, JobContext context)
102           throws WorkflowOperationException {
103 
104     MediaPackage mediaPackage = workflowInstance.getMediaPackage();
105     logger.info("Starting subtitle timeshift workflow for mediapackage: {}", mediaPackage.getIdentifier().toString());
106 
107     // get flavor from workflow configuration
108     final WorkflowOperationInstance operation = workflowInstance.getCurrentOperation();
109     MediaPackageElementFlavor configuredSubtitleFlavor;
110     String configuredVideoFlavors;
111     MediaPackageElementFlavor configuredTargetFlavor;
112     try {
113       configuredSubtitleFlavor = MediaPackageElementFlavor.parseFlavor(
114           Objects.toString(operation.getConfiguration(SUBTITLE_SOURCE_FLAVOR_CFG_KEY)));
115       configuredTargetFlavor = MediaPackageElementFlavor.parseFlavor(
116           operation.getConfiguration(TARGET_FLAVOR_CFG_KEY));
117       configuredVideoFlavors = StringUtils
118           .trimToNull(operation.getConfiguration(VIDEO_SOURCE_FLAVOR_CFG_KEY));
119       if (configuredVideoFlavors == null) {
120         throw new WorkflowOperationException(format("Configuration property %s not set", VIDEO_SOURCE_FLAVOR_CFG_KEY));
121       }
122     } catch (Exception e) {
123       throw new WorkflowOperationException("Couldn't parse subtitle-timeshift workflow configurations.", e);
124     }
125 
126     // In this block we try to get the subtitle and video track for this workflow
127     Track[] originalSubtitleTracks;
128     Track videoTrack;
129     long totalVideoDuration = 0;
130     try {
131       // Get the subtitles and videos from the mediapackage
132       Track[] subtitleTracks = mediaPackage.getTracks(configuredSubtitleFlavor);
133 
134       TrackSelector trackSelector = new TrackSelector();
135       for (String flavor : asList(configuredVideoFlavors)) {
136         trackSelector.addFlavor(flavor);
137       }
138       Collection<Track> videoTracks = trackSelector.select(mediaPackage, false);
139       if (videoTracks.isEmpty()) {
140         throw new WorkflowOperationException(format("No video tracks found in mediapackage %s with flavor %s",
141             mediaPackage.getIdentifier().toString(), configuredVideoFlavors));
142       }
143 
144       // Check if we found the right amount of subtitles and videos
145       // Allowed are exactly 1 video track and at least 1 subtitle track
146       if (subtitleTracks.length == 0) {
147         // if no subtitle track was found, we skip the workflow operation
148         logger.info("No subtitle track found with flavor {}. Skipping subtitle-timeshift workflow operation "
149             + "for mediapackage {}", configuredSubtitleFlavor, mediaPackage.getIdentifier());
150         return createResult(mediaPackage, Action.SKIP);
151       }
152 
153       // these subtitle tracks will be used to create the new subtitle tracks with the shifted timestamps
154       originalSubtitleTracks = subtitleTracks;
155 
156       // this video track will be used to determine how much the subtitle tracks should be shifted
157       for (Track track : videoTracks) {
158         Long duration = track.getDuration();
159         if (duration != null) {
160           totalVideoDuration += duration;
161         } else {
162           logger.debug("Videotrack {} did not have a duration.", track);
163         }
164       }
165 
166       logger.info("Valid tracks found. Start shifting subtitle tracks by duration '{}'", totalVideoDuration);
167 
168     } catch (Exception e) {
169       logger.error("Error in subtitle-timeshift workflow while getting tracks for mediapackage {}",
170           mediaPackage.getIdentifier(), e);
171       throw new WorkflowOperationException(e);
172     }
173 
174     // In this block we try to create the new subtitle tracks and add them to the mediapackage
175     try {
176       for (Track originalSubtitleTrack : originalSubtitleTracks) {
177 
178         // load the subtitle file from workspace and parse it into a webvtt object
179         WebVTTSubtitle newSubtitleFile = loadAndParseSubtitleFile(originalSubtitleTrack);
180 
181         // shift the timestamps of the parsed webvtt object
182         shiftTime(newSubtitleFile, totalVideoDuration);
183 
184         // save the new subtitle file in the workspace to get a URI
185         String originalFileName = FilenameUtils.getBaseName(originalSubtitleTrack.getLogicalName());
186         String newFileName = "timeshifted-" + originalFileName + ".vtt";
187         URI newSubtitleFileUri = saveSubtitleFileToWorkspace(newSubtitleFile, newFileName);
188 
189         // create a track object out of the subtitle URI
190         Track newSubtitleTrack = createNewTrackFromSubtitleUri(newSubtitleFileUri, configuredTargetFlavor,
191             originalSubtitleTrack);
192 
193         // save the new subtitle track to the mediapackage
194         mediaPackage.add(newSubtitleTrack);
195         logger.info("Added subtitle track with URI {} to mediapackage {}", newSubtitleFileUri,
196             mediaPackage.getIdentifier());
197       }
198 
199     } catch (Exception e) {
200       logger.error("Error while shifting time of subtitle tracks for mediapackage {}", mediaPackage.getIdentifier(), e);
201       throw new WorkflowOperationException(e);
202     }
203 
204     logger.info("Subtitle-Timeshift workflow operation for media package {} completed", mediaPackage);
205     return createResult(mediaPackage, Action.CONTINUE);
206   }

208   /**
209    * Takes several parameter for the new subtitle file in and creates a track object from it.
210    *
211    * @param subtitleFile The new subtitle file that will be contained in the track.
212    * @param targetFlavor The future flavor of the new subtitle track.
213    * @param originalSubtitleTrack The original subtitle track.
214    * @return The subtitle file as a track.
215    */
216   private Track createNewTrackFromSubtitleUri(URI subtitleFile, MediaPackageElementFlavor targetFlavor,
217       Track originalSubtitleTrack) throws IOException, NotFoundException {
218 
219     Track newSubtitleTrack = (Track) originalSubtitleTrack.clone();
220     newSubtitleTrack.generateIdentifier();
221     newSubtitleTrack.setFlavor(targetFlavor);
222     newSubtitleTrack.setURI(subtitleFile);
223     newSubtitleTrack.setChecksum(Checksum.create(ChecksumType.DEFAULT_TYPE, workspace.get(subtitleFile, true)));
224     return newSubtitleTrack;
225   }
226 
227   /**
228    * Saves the subtitle object into the workspace and creates a file there.
229    *
230    * @param webVTTSubtitle The subtitle object.
231    * @param fileName The filname of the new subtitle file.
232    * @return The URI of the new subtitle file.
233    * @throws WorkflowOperationException when something went wrong in the parsing and saving process.
234    */
235   private URI saveSubtitleFileToWorkspace(WebVTTSubtitle webVTTSubtitle, String fileName)
236           throws WorkflowOperationException {
237     try (ByteArrayOutputStream outputStream = new ByteArrayOutputStream()) {
238       WebVTTWriter writer = new WebVTTWriter();
239       writer.write(webVTTSubtitle, outputStream);
240       try (ByteArrayInputStream inputStream = new ByteArrayInputStream(outputStream.toByteArray())) {
241         return workspace.putInCollection(COLLECTION, fileName, inputStream);
242       }
243     } catch (IOException e) {
244       logger.error("An exception occurred while parsing and saving a subtitle file to the workspace", e);
245       throw new WorkflowOperationException(e);
246     }
247   }
248 
249   /**
250    * Loads a subtitle file from the workspace and parses it into a WebVTTSubtitle Object
251    *
252    * @param subtitleTrack The track we want to load
253    * @return The parsed webVTTSubtitle object
254    * @throws WorkflowOperationException when something went wrong in the parsing and loading process
255    */
256   private WebVTTSubtitle loadAndParseSubtitleFile(Track subtitleTrack) throws WorkflowOperationException {
257     // Get the subtitle file from workspace
258     File subtitleFile;
259     try {
260       subtitleFile = workspace.get(subtitleTrack.getURI());
261     } catch (IOException ex) {
262       throw new WorkflowOperationException("Can't read " + subtitleTrack.getURI());
263     } catch (NotFoundException ex) {
264       throw new WorkflowOperationException("Workspace does not contain a track " + subtitleTrack.getURI());
265     }
266 
267     // Next try to parse the file into a WebVTT Object
268     WebVTTSubtitle subtitle;
269     try (FileInputStream fin = new FileInputStream(subtitleFile)) {
270       subtitle = new WebVTTParser().parse(fin);
271     } catch (Exception e) {
272       throw new WorkflowOperationException("Couldn't parse subtitle file " + subtitleTrack.getURI(), e);
273     }
274 
275     return subtitle;
276   }

278   /**
279    * Shifts all timestamps of a subtitle file by a given time.
280    *
281    * @param time Time in milliseconds by which all timestamps shall be shifted.
282    */
283   public void shiftTime(WebVTTSubtitle subtitleFile, long time) {
284     for (WebVTTSubtitleCue cue : subtitleFile.getCues()) {
285       long start = cue.getStartTime();
286       long end = cue.getEndTime();
287       cue.setStartTime(start + time);
288       cue.setEndTime(end + time);
289     }
290   }
