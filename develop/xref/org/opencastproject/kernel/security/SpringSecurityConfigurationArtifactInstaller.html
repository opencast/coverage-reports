58  /**
59   * Registers a security filter, which delegates to the spring filter chain appropriate for the current request's
60   * organization. Organizational security configurations may be added to the security watch directory, and should be
61   * named &lt;organization_id&gt;.xml.
62   */
63  @Component(
64      immediate = true,
65      service = ArtifactInstaller.class,
66      property = {
67          "service.description=Security Configuration Scanner"
68      }
69  )
70  public class SpringSecurityConfigurationArtifactInstaller implements ArtifactInstaller {
71    protected static final Logger logger = LoggerFactory.getLogger(SpringSecurityConfigurationArtifactInstaller.class);

79    protected SecurityService securityService = null;

81    protected UserDirectoryService userDirectory = null;
82    /** The user detail service reference for Spring beans */
83    protected UserDetailsService userDetailsService = null;
84    /** The user reference provider service reference for Spring beans */
85    protected UserReferenceProvider userReferenceProvider = null;
86    /** LTI 1.1. configuration services */
87    protected OAuthConsumerDetailsService oAuthConsumerDetailsService = null;
88    protected LtiLaunchAuthenticationHandler ltiLaunchAuthenticationHandler = null;
89    /** LDAP authorities populators, keyed by organization id and, inside each organization, keyed by pid/instanceId */
90    protected Map<String, Map<String, LdapAuthoritiesPopulator>> ldapAuthoritiesPopulators =
91            new HashMap<String, Map<String, LdapAuthoritiesPopulator>>();
92    /** List of expected LDAP instances for each organization */
93    private Map<String, Set<String>> expectedLdapInstances = new HashMap<String, Set<String>>();
94    /**
95     * List of security/ORG_ID.xml files that were waiting for LDAP authorities populator instances, keyed by organization
96     * ID
97     */
98    private Map<String, File> pendingArtifactsToInstall = new HashMap<String, File>();

157   /* These are set as LdapAuthoritiesPopulator service properties when they are registered */
158   private static final String INSTANCE_ID_SERVICE_PROPERTY_KEY = "instanceId";
159   private static final String ORGANIZATION_ID_SERVICE_PROPERTY_KEY = "orgId";
160 
161   @Reference(
162       cardinality = ReferenceCardinality.MULTIPLE, // 0..n
163           policy = ReferencePolicy.DYNAMIC, unbind = "unbindLdapAuthoritiesPopulator"
164   )
165   synchronized void bindLdapAuthoritiesPopulator(LdapAuthoritiesPopulator service, Map<String, Object> properties)
166           throws Exception {
167     String orgId = (String) properties.get(ORGANIZATION_ID_SERVICE_PROPERTY_KEY);
168     String instanceId = (String) properties.get(INSTANCE_ID_SERVICE_PROPERTY_KEY);
169     if (ldapAuthoritiesPopulators.get(orgId) == null) {
170       ldapAuthoritiesPopulators.put(orgId, new HashMap<String, LdapAuthoritiesPopulator>());
171     }
172     ldapAuthoritiesPopulators.get(orgId).put(instanceId, service);
173 
174     // If all expected LDAP instances ready, process pending artifact if there is one
175     if (allExpectedAlreadyRegistered(orgId) && pendingArtifactsToInstall.containsKey(orgId)) {
176       createSecurityContext(pendingArtifactsToInstall.get(orgId));
177       // Remove from pending list
178       pendingArtifactsToInstall.remove(orgId);
179     }
180   }
181 
182   synchronized void unbindLdapAuthoritiesPopulator(LdapAuthoritiesPopulator service, Map<String, Object> properties) {
183     String orgId = (String) properties.get(ORGANIZATION_ID_SERVICE_PROPERTY_KEY);
184     String instanceId = (String) properties.get(INSTANCE_ID_SERVICE_PROPERTY_KEY);
185     if (ldapAuthoritiesPopulators.get(orgId) != null) {
186       ldapAuthoritiesPopulators.get(orgId).remove(instanceId);
187     }
188   }

190   /**
191    * OSGI activation callback
192    */
193   @Activate
194   protected void activate(ComponentContext cc) {
195     this.bundleContext = cc.getBundleContext();
196     this.appContexts = new HashMap<>();
197 
198     Collections.list(cc.getProperties().keys()).stream().filter(s -> s.startsWith(LDAP_INSTANCES_PREFIX)).forEach(s -> {
199       String orgId = s.substring(LDAP_INSTANCES_PREFIX.length());
200       String instanceList = OsgiUtil.getComponentContextProperty(cc, s);
201       expectedLdapInstances.put(orgId, new HashSet(Arrays.asList(instanceList.split(","))));
202     });
203     expectedLdapInstances.forEach((orgId, instances) -> logger.info(orgId + " = " + instances));
204   }

206   /**
207    * {@inheritDoc}

209    * @see org.apache.felix.fileinstall.ArtifactListener#canHandle(java.io.File)

211   @Override
212   public boolean canHandle(File artifact) {
213     return "security".equals(artifact.getParentFile().getName()) && artifact.getName().endsWith(".xml");
214   }

216   /**
217    * {@inheritDoc}
218    *
219    * @see org.apache.felix.fileinstall.ArtifactInstaller#install(java.io.File)
220    */
221   @Override
222   public synchronized void install(File artifact) throws Exception {
223     logger.trace("Install " + artifact.getAbsolutePath());
224 
225     // Are there any expected LDAP authorities populators and have all already registered?
226     String orgId = FilenameUtils.getBaseName(artifact.getName());
227     if (allExpectedAlreadyRegistered(orgId)) {
228       createSecurityContext(artifact);
229     } else {
230       logger.warn("Not all LDAP authorities populators registered so artifact installation is pending.");
231       pendingArtifactsToInstall.put(orgId, artifact);
232     }
233   }
234 
235   /**
236    * Returns true if all expected LDAP authorities populators have already registered.
237    *
238    * @param orgId
239    *          The organization ID
240    * @return true if all registered
241    */
242   private boolean allExpectedAlreadyRegistered(String orgId) {
243     Set<String> expected = expectedLdapInstances.get(orgId) != null
244             ? expectedLdapInstances.get(orgId)
245             : new HashSet<String>();
246     Set<String> registered = ldapAuthoritiesPopulators.get(orgId) != null
247             ? ldapAuthoritiesPopulators.get(orgId).keySet()
248             : new HashSet<String>();
249 
250     return expected.size() == 0 || registered.containsAll(expected);
251   }

253   private void createSecurityContext(File artifact) throws Exception {
254     // If we already have a registration for this ID, take it out of the security filter and close it
255     String orgId = FilenameUtils.getBaseName(artifact.getName());
256     logger.info("Creating security context for organization {}", orgId);
257 
258     GenericApplicationContext orgAppContext = appContexts.get(orgId);
259     if (orgAppContext != null) {
260       securityFilter.removeFilter(orgId);
261       orgAppContext.close();
262     }
263 
264     orgAppContext = new GenericApplicationContext();
265     // this did NOT work orgAppContext.setClassLoader(SecurityNamespaceHandler.class.getClassLoader());
266     // When loading the beans, setting the spring application context class
267     // loader to this bundle's and explicitly importing the spring classes in the pom fixed
268     // the ClassNotFoundExceptions. TODO Is this the best way? Is there more to add?
269     orgAppContext.setClassLoader(this.getClass().getClassLoader());
270 
271     // Manually add the OSGI dependencies
272     orgAppContext.getBeanFactory().registerSingleton("securityService", this.securityService);
273     orgAppContext.getBeanFactory().registerSingleton("userDirectoryService", this.userDirectory);
274     orgAppContext.getBeanFactory().registerSingleton("userDetailsService", this.userDetailsService);
275     orgAppContext.getBeanFactory().registerSingleton("userReferenceProvider", this.userReferenceProvider);
276     orgAppContext.getBeanFactory().registerSingleton("oAuthConsumerDetailsService", this.oAuthConsumerDetailsService);
277     orgAppContext.getBeanFactory().registerSingleton("ltiLaunchAuthenticationHandler",
278             this.ltiLaunchAuthenticationHandler);
279     if (ldapAuthoritiesPopulators.get(orgId) != null) {
280       for (Map.Entry<String, LdapAuthoritiesPopulator> entry : ldapAuthoritiesPopulators.get(orgId).entrySet()) {
281         logger.trace("Registering bean {}", LDAP_AUTH_POPULATOR_BEAN_NAME_PREFIX + entry.getKey());
282         orgAppContext.getBeanFactory().registerSingleton(LDAP_AUTH_POPULATOR_BEAN_NAME_PREFIX + entry.getKey(),
283                 entry.getValue());
284       }
285     }
286 
287     XmlBeanDefinitionReader xmlBeanDefinitionReader = new XmlBeanDefinitionReader(orgAppContext);
288     // So that it finds META-INF/spring.handlers
289     xmlBeanDefinitionReader.setNamespaceHandlerResolver(
290             new DefaultNamespaceHandlerResolver(SecurityNamespaceHandler.class.getClassLoader()));
291     FileSystemResource beanFile = new FileSystemResource(artifact);
292     xmlBeanDefinitionReader.loadBeanDefinitions(beanFile);
293 
294     logger.info("registered {} items in {} for {} from file {}", orgAppContext.getBeanDefinitionCount(),
295             orgAppContext.getBeanDefinitionNames(), orgId, artifact.getAbsolutePath());
296 
297     // Refresh the spring application context
298     try {
299       orgAppContext.refresh();
300     } catch (Exception e) {
301       logger.error("Unable to refresh spring security configuration file {}: {}", artifact, e);
302       orgAppContext.close();
303       return;
304     }
305 
306     // Keep track of the app context so we can close it later
307     appContexts.put(orgId, orgAppContext);
308 
309     // Add the filter chain for this org to the security filter
310     securityFilter.addFilter(orgId, (Filter) orgAppContext.getBean("springSecurityFilterChain"));
311   }

313   /**
314    * {@inheritDoc}
315    *
316    * @see org.apache.felix.fileinstall.ArtifactInstaller#uninstall(java.io.File)
317    */
318   @Override
319   public void uninstall(File artifact) throws Exception {
320     String orgId = FilenameUtils.getBaseName(artifact.getName());
321     GenericApplicationContext appContext = appContexts.get(orgId);
322     if (appContext != null) {
323       securityFilter.removeFilter(orgId);
324       appContexts.remove(orgId);
325       appContext.close();
326     }
327   }
