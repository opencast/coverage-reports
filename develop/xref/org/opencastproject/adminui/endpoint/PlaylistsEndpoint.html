180   @Path("{id}")
181   @RestQuery(
182       name = "getPlaylist",
183       description = "Get a playlist by ID.",
184       returnDescription = "A playlist as JSON",
185       pathParameters = {
186           @RestParameter(name = "id", isRequired = true, description = "The playlist identifier", type = STRING)
187       },
188       responses = {
189           @RestResponse(description = "Returns the playlist.", responseCode = SC_OK),
190           @RestResponse(description = "The specified playlist does not exist.",
191               responseCode = SC_NOT_FOUND),
192           @RestResponse(description = "The user does not have permission to access this playlist.",
193               responseCode = SC_FORBIDDEN),
194           @RestResponse(description = "The request is invalid or inconsistent.",
195               responseCode = SC_BAD_REQUEST)
196       })
197   public Response getPlaylist(
198       @HeaderParam("Accept") String acceptHeader,
199       @PathParam("id") String id) {
200     try {
201       Playlist playlist = service.getPlaylistById(id);
202       return Response.ok(playlistToJson(playlist).toString(), MediaType.APPLICATION_JSON_TYPE).build();
203     } catch (NotFoundException e) {
204       logger.info("Playlist '{}' not found.", id);
205       return Response.status(Response.Status.NOT_FOUND).build();
206     } catch (UnauthorizedException e) {
207       logger.warn("User doesn't have permission to access playlist '{}'.", id);
208       return Response.status(Response.Status.FORBIDDEN).build();
209     } catch (IllegalStateException e) {
210       logger.warn("Invalid state when accessing playlist '{}': {}", id, e.getMessage());
211       return Response.status(Response.Status.BAD_REQUEST).build();
212     }
213   }

215   @GET
216   @Path("")
217   @RestQuery(
218       name = "getPlaylists",
219       description = "Get playlists. Playlists that you do not have read access to will not show up.",
220       returnDescription = "A JSON object containing an array of playlists.",
221       restParameters = {
222           @RestParameter(name = "limit", isRequired = false, type = INTEGER,
223               description = "The maximum number of results to return for a single request.", defaultValue = "100"),
224           @RestParameter(name = "offset", isRequired = false, type = INTEGER,
225               description = "The index of the first result to return.", defaultValue = "0"),
226           @RestParameter(name = "sort", isRequired = false, type = STRING,
227               description = "Sort the results based upon a sorting criteria. Must be in the form '<Sort "
228                   + "Name>:ASC' or '<Sort Name>:DESC'. Sort Name is case sensitive."
229                   + " Supported Sort Names are 'updated', 'title', and 'creator'.",
230               defaultValue = "updated:ASC"),
231           @RestParameter(name = "filter", isRequired = false, type = STRING,
232               description = "Filter the results by the given criteria. The format is 'key:value', "
233                   + "and multiple filters can be combined with commas. Supported filter keys: "
234                   + "'textFilter' (case-insensitive search across title, description, and creator), "
235                   + "'creator' (case-insensitive match on the creator field), "
236                   + "'Updated' (date range in the form 'start/end' using ISO 8601 date-time values).",
237               defaultValue = "textFilter:opencast,creator:john,Updated:2025-01-01/2026-12-31")
238       },
239       responses = {
240           @RestResponse(description = "Returns the playlists.", responseCode = SC_OK),
241           @RestResponse(description = "The request is invalid or inconsistent.",
242               responseCode = SC_BAD_REQUEST)
243       })
244   public Response getPlaylists(
245       @HeaderParam("Accept") String acceptHeader,
246       @QueryParam("limit") Integer limit,
247       @QueryParam("offset") Integer offset,
248       @QueryParam("sort") String sort,
249       @QueryParam("filter") String filter) {
250 
251     Optional<Integer> optLimit = Optional.ofNullable(limit);
252     Optional<Integer> optOffset = Optional.ofNullable(offset);
253     Optional<String> optSort = Optional.ofNullable(trimToNull(sort));
254 
255     // If the limit is set to 0, this is not taken into account.
256     if (optLimit.isPresent() && limit == 0) {
257       optLimit = Optional.empty();
258     }
259 
260     List<Predicate<Playlist>> filterPredicates = new ArrayList<>();
261 
262     // Add filters
263     Map<String, String> filters = RestUtils.parseFilter(filter);
264     for (String name : filters.keySet()) {
265       String value = filters.get(name);
266       if (PlaylistsListQuery.FILTER_TEXT_NAME.equals(name)) {
267         filterPredicates.add(containsIgnoreCase(Playlist::getTitle, value)
268             .or(containsIgnoreCase(Playlist::getDescription, value))
269             .or(containsIgnoreCase(Playlist::getCreator, value)));
270       } else if (PlaylistsListQuery.FILTER_CREATOR_NAME.equals(name)) {
271         filterPredicates.add(containsIgnoreCase(Playlist::getCreator, value));
272       } else if (PlaylistsListQuery.FILTER_UPDATED_NAME.equals(name)) {
273         try {
274           Tuple<Date, Date> range = RestUtils.getFromAndToDateRange(value);
275           filterPredicates.add(p -> p.getUpdated() != null
276               && !p.getUpdated().before(range.getA())
277               && !p.getUpdated().after(range.getB()));
278         } catch (IllegalArgumentException e) {
279           logger.warn("Could not parse Updated filter dates: {}", value);
280         }
281       } else {
282         logger.debug("Unknown filter: {}", name);
283       }
284     }
285 
286     SortCriterion sortCriterion
287         = new SortCriterion("updated", SortCriterion.Order.Ascending);
288     if (optSort.isPresent()) {
289       ArrayList<SortCriterion> sortCriteria
290           = RestUtils.parseSortQueryParameter(optSort.get());
291       for (SortCriterion criterion : sortCriteria) {
292         switch (criterion.getFieldName()) {
293           case "updated":
294           case "creator":
295           case "title":
296             sortCriterion = criterion;
297             break;
298           default:
299             logger.warn("Unknown sort criteria: {}",
300                 criterion.getFieldName());
301             return Response.status(Response.Status.BAD_REQUEST).build();
302         }
303       }
304     }
305 
306     // Fetch all playlists (filtering is done in memory since
307     // playlists are not in the search index). Todo: Add to index?
308     List<Playlist> allPlaylists
309         = service.getPlaylists(Integer.MAX_VALUE, 0, sortCriterion);
310 
311     // Apply filters
312     List<Playlist> filteredPlaylists = allPlaylists;
313     for (Predicate<Playlist> predicate : filterPredicates) {
314       filteredPlaylists = filteredPlaylists.stream()
315           .filter(predicate).toList();
316     }
317 
318     // Apply pagination
319     int effectiveOffset = optOffset.orElse(0);
320     int effectiveLimit = optLimit.orElse(DEFAULT_LIMIT);
321     int actualOffset = Math.min(effectiveOffset, filteredPlaylists.size());
322     int endIndex = Math.min(
323         actualOffset + effectiveLimit, filteredPlaylists.size());
324     List<Playlist> paginatedPlaylists
325         = filteredPlaylists.subList(actualOffset, endIndex);
326 
327     List<JsonObject> playlistsList = new ArrayList<>();
328     for (Playlist p : paginatedPlaylists) {
329       playlistsList.add(playlistToJson(p));
330     }
331 
332     return okJsonList(playlistsList, effectiveOffset, effectiveLimit,
333         filteredPlaylists.size());
334   }

337   @Path("")
338   @RestQuery(
339       name = "createPlaylist",
340       description = "Creates a new playlist.",
341       returnDescription = "The created playlist.",
342       restParameters = {
343           @RestParameter(name = "playlist", isRequired = true, description = "Playlist in JSON format", type = TEXT,
344               defaultValue = SAMPLE_PLAYLIST_JSON)
345       },
346       responses = {
347           @RestResponse(description = "Playlist created.", responseCode = SC_CREATED),
348           @RestResponse(description = "The user does not have permission to create a playlist.",
349               responseCode = SC_FORBIDDEN),
350           @RestResponse(description = "The request is invalid or inconsistent.",
351               responseCode = SC_BAD_REQUEST)
352       })
353   public Response createPlaylist(
354       @HeaderParam("Accept") String acceptHeader,
355       @FormParam("playlist") String playlistText) {
356     try {
357       // Parse JSON to playlist
358       Playlist playlist = restService.parseJsonToPlaylist(playlistText);
359 
360       // Persist
361       playlist = service.update(playlist);
362 
363       return Response.created(URI.create(getPlaylistUrl(playlist.getId())))
364           .entity(playlistToJson(playlist).toString())
365           .type(MediaType.APPLICATION_JSON_TYPE)
366           .build();
367     } catch (UnauthorizedException e) {
368       logger.warn("User doesn't have permission to create playlist.");
369       return Response.status(Response.Status.FORBIDDEN).build();
370     } catch (ParseException | IOException | IllegalArgumentException e) {
371       logger.warn("Invalid playlist JSON: {}", e.getMessage());
372       return Response.status(Response.Status.BAD_REQUEST).build();
373     }
374   }

443   private JsonObject playlistToJson(Playlist playlist) {
444     JsonObject json = new JsonObject();
445 
446     json.addProperty("id", playlist.getId());
447 
448     // Look up event metadata for all entries so the frontend doesn't have to
449     Map<String, Event> eventMap = lookupEntryEvents(playlist.getEntries());
450 
451     JsonArray entriesArray = new JsonArray();
452     for (PlaylistEntry entry : playlist.getEntries()) {
453       entriesArray.add(playlistEntryToJson(entry, eventMap.get(entry.getContentId())));
454     }
455     json.add("entries", entriesArray);
456 
457     json.addProperty("title", safeString(playlist.getTitle()));
458     json.addProperty("description", safeString(playlist.getDescription()));
459     json.addProperty("creator", safeString(playlist.getCreator()));
460     json.addProperty("updated", playlist.getUpdated() != null ? toUTC(playlist.getUpdated().getTime()) : "");
461 
462     JsonArray aceArray = new JsonArray();
463     for (PlaylistAccessControlEntry ace : playlist.getAccessControlEntries()) {
464       aceArray.add(playlistAccessControlEntryToJson(ace));
465     }
466     json.add("accessControlEntries", aceArray);
467 
468     return json;
469   }

471   private JsonObject playlistEntryToJson(PlaylistEntry playlistEntry, Event event) {
472     JsonObject json = new JsonObject();
473 
474     json.addProperty("id", playlistEntry.getId());
475     if (playlistEntry.getContentId() != null) {
476       json.addProperty("contentId", playlistEntry.getContentId());
477     } else {
478       json.add("contentId", null);
479     }
480 
481     json.add("type", enumToJSON(playlistEntry.getType()));
482 
483     // Include event metadata if available
484     if (event != null) {
485       json.addProperty("title", safeString(event.getTitle()));
486       json.addProperty("start_date", safeString(event.getRecordingStartDate()));
487 
488       if (event.getSeriesName() != null) {
489         JsonObject series = new JsonObject();
490         series.addProperty("id", safeString(event.getSeriesId()));
491         series.addProperty("title", safeString(event.getSeriesName()));
492         json.add("series", series);
493       }
494 
495       if (event.getPresenters() != null && !event.getPresenters().isEmpty()) {
496         JsonArray presenters = new JsonArray();
497         for (String presenter : event.getPresenters()) {
498           presenters.add(presenter);
499         }
500         json.add("presenters", presenters);
501       }
502     }
503 
504     return json;
505   }

507   /**
508    * Look up event metadata from search index for all playlist entries.
509    */
510   private Map<String, Event> lookupEntryEvents(List<PlaylistEntry> entries) {
511     Map<String, Event> eventMap = new HashMap<>();
512     String org = securityService.getOrganization().getId();
513     var user = securityService.getUser();
514 
515     for (PlaylistEntry entry : entries) {
516       String contentId = entry.getContentId();
517       if (contentId == null || contentId.isEmpty()) {
518         continue;
519       }
520 
521       try {
522         SearchResult<Event> result = elasticsearchIndex.getByQuery(
523             new EventSearchQuery(org, user).withIdentifier(contentId));
524         if (result.getPageSize() != 0) {
525           eventMap.put(contentId, result.getItems()[0].getSource());
526         }
527       } catch (SearchIndexException e) {
528         logger.warn("Could not look up event '{}': {}", contentId, e.getMessage());
529       }
530     }
531 
532     return eventMap;
533   }

554   /** Build a case-insensitive contains predicate for a string field of a playlist. */
555   private static Predicate<Playlist> containsIgnoreCase(Function<Playlist, String> getter, String value) {
556     String lower = value.toLowerCase();
557     return p -> {
558       String field = getter.apply(p);
559       return field != null && field.toLowerCase().contains(lower);
560     };
