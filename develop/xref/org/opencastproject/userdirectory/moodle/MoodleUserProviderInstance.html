224   @Activate
225   @Modified
226   public void updated(BundleContext bundleContext, Map<String, Object> properties) throws ConfigurationException {
227     String adminUserName = StringUtils.trimToNull(
228         bundleContext.getProperty(SecurityConstants.GLOBAL_ADMIN_USER_PROPERTY)
229     );
230 
231     String organizationId = (String) properties.get(ORGANIZATION_KEY);
232     if (StringUtils.isBlank(organizationId)) {
233       throw new ConfigurationException(ORGANIZATION_KEY, "is not set");
234     }
235 
236     String urlStr = (String) properties.get(MOODLE_URL_KEY);
237     URI url;
238     if (StringUtils.isBlank(urlStr)) {
239       throw new ConfigurationException(MOODLE_URL_KEY, "is not set");
240     }
241     try {
242       url = new URI(urlStr);
243     } catch (URISyntaxException e) {
244       throw new ConfigurationException(MOODLE_URL_KEY, "not a URL");
245     }
246 
247     String token = (String) properties.get(MOODLE_TOKEN_KEY);
248     if (StringUtils.isBlank(token)) {
249       throw new ConfigurationException(MOODLE_TOKEN_KEY, "is not set");
250     }
251 
252     groupRoles = BooleanUtils.toBoolean((String) properties.get(GROUP_ROLES_KEY));
253 
254     coursePattern = (String) properties.get(COURSE_PATTERN_KEY);
255     userPattern = (String) properties.get(USER_PATTERN_KEY);
256     groupPattern = (String) properties.get(GROUP_PATTERN_KEY);
257     lowercaseUsername = BooleanUtils.toBoolean((String) properties.get(LOWERCASE_USERNAME));
258 
259     contextRolePrefix = Objects.toString(properties.get(CONTEXT_ROLE_PREFIX), "");
260 
261     int cacheSize = 1000;
262     try {
263       if (properties.get(CACHE_SIZE) != null) {
264         cacheSize = Integer.parseInt(properties.get(CACHE_SIZE).toString());
265       }
266     } catch (NumberFormatException e) {
267       logger.warn("{} could not be loaded, default value is used: {}", CACHE_SIZE, cacheSize);
268     }
269 
270     int cacheExpiration = 60;
271     try {
272       if (properties.get(CACHE_EXPIRATION) != null) {
273         cacheExpiration = Integer.parseInt(properties.get(CACHE_EXPIRATION).toString());
274       }
275     } catch (NumberFormatException e) {
276       logger.warn("{} could not be loaded, default value is used: {}", CACHE_EXPIRATION, cacheExpiration);
277     }
278 
279     try {
280       organization = orgDirectory.getOrganization(organizationId);
281     } catch (NotFoundException e) {
282       logger.warn("Organization {} not found!", organizationId);
283       throw new ConfigurationException(ORGANIZATION_KEY, "not found");
284     }
285 
286     client = new MoodleWebServiceImpl(url, token);
287 
288     // initialize user filter
289     ignoredUsernames = new ArrayList<>();
290     ignoredUsernames.add("");
291     ignoredUsernames.add(SecurityConstants.GLOBAL_ANONYMOUS_USERNAME);
292     if (StringUtils.isNoneEmpty(adminUserName)) {
293       ignoredUsernames.add(adminUserName);
294     }
295 
296     logger.info("Configured MoodleUserProviderInstance(url={}, cacheSize={}, cacheExpiration={})",
297             client.getURL(), cacheSize, cacheExpiration);
298 
299     // Setup the caches
300     cache = CacheBuilder.newBuilder().maximumSize(cacheSize).expireAfterWrite(cacheExpiration, TimeUnit.MINUTES)
301             .build(new CacheLoader<String, Object>() {
302               @Override
303               public Object load(String username) {
304                 User user = loadUserFromMoodle(username);
305                 return user == null ? nullToken : user;
306               }
307             });
308   }

334   /**
335    * {@inheritDoc}
336    *
337    * @see org.opencastproject.security.api.UserProvider#loadUser(java.lang.String)
338    */
339   @Override
340   public User loadUser(String userName) {
341     try {
342       Object user = cache.getUnchecked(userName);
343       if (user == nullToken) {
344         logger.debug("Returning null user from cache");
345         return null;
346       } else {
347         logger.debug("Returning user {} from cache", userName);
348         return (User) user;
349       }
350     } catch (ExecutionError e) {
351       logger.warn("Exception while loading user {}", userName, e);
352       return null;
353     } catch (UncheckedExecutionException e) {
354       logger.warn("Exception while loading user {}", userName, e);
355       return null;
356     }
357   }

380   /**
381    * {@inheritDoc}
382    *
383    * @see org.opencastproject.security.api.UserProvider#findUsers(java.lang.String, int, int)
384    */
385   @Override
386   public Iterator<User> findUsers(String query, int offset, int limit) {
387     if (query == null) {
388       throw new IllegalArgumentException("Query must be set");
389     }
390 
391     if (query.endsWith("%")) {
392       query = query.substring(0, query.length() - 1);
393     }
394 
395     if (query.isEmpty()) {
396       return Collections.emptyIterator();
397     }
398 
399     // Check if user matches pattern
400     try {
401       if ((userPattern != null) && !query.matches(userPattern)) {
402         logger.debug("verify user {} failed regexp {}", query, userPattern);
403         return Collections.emptyIterator();
404       }
405     } catch (PatternSyntaxException e) {
406       logger.warn("Invalid regular expression for user pattern {} - disabling checks", userPattern);
407       userPattern = null;
408     }
409 
410     // Load User
411     List<User> users = new LinkedList<>();
412 
413     User user = loadUser(query);
414     if (user != null) {
415       users.add(user);
416     }
417 
418     return users.iterator();
419   }

434   /**
435    * {@inheritDoc}
436    *
437    * @see org.opencastproject.security.api.RoleProvider#getRolesForUser(java.lang.String)
438    */
439   @Override
440   public List<Role> getRolesForUser(String username) {
441     List<Role> roles = new LinkedList<>();
442 
443     // Don't answer for admin, anonymous or empty user
444     if (ignoredUsernames.stream().anyMatch(u -> u.equals(username))) {
445       logger.debug("we don't answer for: {}", username);
446       return roles;
447     }
448 
449     User user = loadUser(username);
450     if (user != null) {
451       logger.debug("Returning cached role set for {}", username);
452       return new ArrayList<>(user.getRoles());
453     }
454 
455     // Not found
456     logger.debug("Return empty role set for {} - not found in Moodle", username);
457     return new LinkedList<>();
458   }

460   /**
461    * {@inheritDoc}
462    * <p>
463    * We search for COURSEID, COURSEID_Learner, COURSEID_Instructor
464    */
465   @Override
466   public Iterator<Role> findRoles(String query, Role.Target target, int offset, int limit) {
467     // Don't return roles for users or groups
468     if (target == Role.Target.USER) {
469       return Collections.emptyIterator();
470     }
471 
472     boolean exact = true;
473     boolean ltirole = false;
474 
475     if (query.endsWith("%")) {
476       exact = false;
477       query = query.substring(0, query.length() - 1);
478     }
479 
480     if (query.isEmpty()) {
481       return Collections.emptyIterator();
482     }
483 
484     // Verify query starts with prefix configured for this user provider instance
485     if (!query.startsWith(contextRolePrefix)) {
486       return Collections.emptyIterator();
487     }
488 
489     // Verify that role name ends with LEARNER_ROLE_SUFFIX or INSTRUCTOR_ROLE_SUFFIX
490     if (exact
491         && !query.endsWith("_" + LEARNER_ROLE_SUFFIX)
492         && !query.endsWith("_" + INSTRUCTOR_ROLE_SUFFIX)
493         && !query.endsWith("_" + GROUP_ROLE_SUFFIX)) {
494       return Collections.emptyIterator();
495     }
496 
497     final String groupRolePrefix = contextRolePrefix + GROUP_ROLE_PREFIX;
498     final boolean findGroupRole = groupRoles && query.startsWith(groupRolePrefix);
499 
500     // Extract Moodle id
501     String moodleId = findGroupRole ? query.substring(groupRolePrefix.length()) : query;
502     if (query.endsWith("_" + LEARNER_ROLE_SUFFIX)) {
503       moodleId = query.substring(contextRolePrefix.length(), query.lastIndexOf("_" + LEARNER_ROLE_SUFFIX));
504       ltirole = true;
505     } else if (query.endsWith("_" + INSTRUCTOR_ROLE_SUFFIX)) {
506       moodleId = query.substring(contextRolePrefix.length(), query.lastIndexOf("_" + INSTRUCTOR_ROLE_SUFFIX));
507       ltirole = true;
508     } else if (query.endsWith("_" + GROUP_ROLE_SUFFIX)) {
509       moodleId = query.substring(contextRolePrefix.length(), query.lastIndexOf("_" + GROUP_ROLE_SUFFIX));
510       ltirole = true;
511     }
512 
513     // Check if matches patterns
514     String pattern = findGroupRole ? groupPattern : coursePattern;
515     try {
516       if ((pattern != null) && !moodleId.matches(pattern)) {
517         logger.debug("Verify Moodle ID {} failed regexp {}", moodleId, pattern);
518         return Collections.emptyIterator();
519       }
520     } catch (PatternSyntaxException e) {
521       logger.warn("Invalid regular expression for pattern {} - disabling checks", pattern);
522       if (findGroupRole) {
523         groupPattern = null;
524       } else {
525         coursePattern = null;
526       }
527     }
528 
529     // Roles list
530     List<Role> roles = new LinkedList<>();
531     JaxbOrganization jaxbOrganization = JaxbOrganization.fromOrganization(organization);
532     if (ltirole) {
533       // Query is for a Moodle ID and an LTI role (Instructor/Learner/Group)
534       roles.add(new JaxbRole(query, jaxbOrganization, "Moodle Site Role", Role.Type.EXTERNAL));
535     } else if (findGroupRole) {
536       // Group ID
537       roles.add(new JaxbRole(contextRolePrefix + GROUP_ROLE_PREFIX + moodleId + "_" + GROUP_ROLE_SUFFIX,
538           jaxbOrganization, "Moodle Group Learner Role", Role.Type.EXTERNAL));
539     } else {
540       // Course ID - return both roles
541       roles.add(new JaxbRole(moodleId + "_" + INSTRUCTOR_ROLE_SUFFIX, jaxbOrganization,
542               "Moodle Course Instructor Role", Role.Type.EXTERNAL));
543       roles.add(new JaxbRole(moodleId + "_" + LEARNER_ROLE_SUFFIX, jaxbOrganization, "Moodle Course Learner Role",
544               Role.Type.EXTERNAL));
545     }
546 
547     return roles.iterator();
548   }

553   /**
554    * Loads a user from Moodle.
555    *
556    * @param username The username.
557    * @return The user.
558    */
559   private User loadUserFromMoodle(String username) {
560     if (lowercaseUsername) {
561       username = username.toLowerCase();
562     }
563 
564     logger.debug("loadUserFromMoodle({})", username);
565 
566     if (cache == null) {
567       throw new IllegalStateException("The Moodle user detail service has not yet been configured");
568     }
569 
570     // Don't answer for admin, anonymous or empty user
571     if (ignoredUsernames.contains(username)) {
572       logger.debug("We don't answer for: " + username);
573       return null;
574     }
575 
576     JaxbOrganization jaxbOrganization = JaxbOrganization.fromOrganization(organization);
577 
578     Thread currentThread = Thread.currentThread();
579     ClassLoader originalClassloader = currentThread.getContextClassLoader();
580 
581     try {
582       // Load user
583       List<MoodleUser> moodleUsers = client
584               .coreUserGetUsersByField(CoreUserGetUserByFieldFilters.username, Collections.singletonList(username));
585 
586       if (moodleUsers.isEmpty()) {
587         logger.debug("User {} not found in Moodle system", username);
588         return null;
589       }
590       MoodleUser moodleUser = moodleUsers.get(0);
591 
592       // Load Roles
593       List<String> courseIdsInstructor = client.toolOpencastGetCoursesForInstructor(username);
594       List<String> courseIdsLearner = client.toolOpencastGetCoursesForLearner(username);
595       List<String> groupIdsLearner = groupRoles
596           ? client.toolOpencastGetGroupsForLearner(username)
597           : Collections.emptyList();
598 
599       // Create Opencast Objects
600       Set<JaxbRole> roles = new HashSet<>();
601       roles.add(new JaxbRole(Group.ROLE_PREFIX + contextRolePrefix + "MOODLE", jaxbOrganization, "Moodle Users",
602           Role.Type.EXTERNAL_GROUP));
603       for (final String courseId : courseIdsInstructor) {
604         roles.add(contextRole(courseId, INSTRUCTOR_ROLE_SUFFIX, jaxbOrganization));
605       }
606       for (final String courseId : courseIdsLearner) {
607         roles.add(contextRole(courseId, LEARNER_ROLE_SUFFIX, jaxbOrganization));
608       }
609       for (final String groupId : groupIdsLearner) {
610         roles.add(contextRole(GROUP_ROLE_PREFIX + groupId, GROUP_ROLE_SUFFIX, jaxbOrganization));
611       }
612 
613       return new JaxbUser(moodleUser.getUsername(), null, moodleUser.getFullname(), moodleUser.getEmail(),
614               this.getName(), jaxbOrganization, roles);
615     } catch (Exception e) {
616       logger.warn("Exception loading Moodle user {} at {}", username, client.getURL());
617     } finally {
618       currentThread.setContextClassLoader(originalClassloader);
619     }
620 
621     return null;
622   }

624   /**
625    * Create an Opencast JaxbRole based on a Moodle user's context.
626    *
627    * @param context Moodle user's context like course identifier
628    * @param contextRole Moodle user's context role like Instructor
629    * @param organization Opencast organization to create user for
630    * @return JaxbRole
631    */
632   private JaxbRole contextRole(final String context, final String contextRole, final JaxbOrganization organization) {
633     final String name = contextRolePrefix + context + "_" + contextRole;
634     final String description = "Moodle Course " + contextRole + " Role";
635     return new JaxbRole(name, organization, description, Role.Type.EXTERNAL);
636   }
