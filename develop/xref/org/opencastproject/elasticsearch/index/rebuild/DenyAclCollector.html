22  package org.opencastproject.elasticsearch.index.rebuild;
23  
24  import java.util.Collections;
25  import java.util.LinkedHashSet;
26  import java.util.Set;
27  
28  /**
29   * Collects the identifiers of objects whose ACL contains deny rules, so that an index rebuild can
30   * report them once at the end instead of logging a line per entry.
31   *
32   * <p>The search index cannot express denial, so such entries are dropped when an object is indexed.
33   * That is worth telling an administrator about, but the per-entry message it used to produce fired
34   * once per deny entry per object and drowned the rebuild log.
35   *
36   * <p>Collection is only active between {@link #start()} and {@link #stop()}, which
37   * {@link IndexRebuildService} brackets a rebuild with. Outside a rebuild — during ordinary
38   * indexing — {@link #record} does nothing beyond a volatile read, so nothing accumulates in a
39   * long-running instance.
40   *
41   * <p>State is held per thread. A rebuild runs on the thread that called it, and keeping the
42   * collected ids thread-local means concurrent ordinary indexing on other threads cannot add to a
43   * rebuild's report, nor can two rebuilds interfere.
44   */
45  public final class DenyAclCollector {
46  
47    /** Ids collected for the rebuild running on this thread, or null when no rebuild is running. */
48    private static final ThreadLocal<Set<String>> COLLECTED = new ThreadLocal<>();
49  
50    private DenyAclCollector() {
51    }
52  
53    /**
54     * Start collecting on the current thread, discarding anything previously collected.
55     */
56    public static void start() {
57      COLLECTED.set(new LinkedHashSet<>());
58    }
59  
60    /**
61     * Note that the object with the given identifier carries at least one deny entry. Does nothing
62     * when no rebuild is running on this thread.
63     *
64     * @param identifier
65     *          the identifier of the event or series, may be null
66     */
67    public static void record(String identifier) {
68      Set<String> collected = COLLECTED.get();
69      if (collected != null && identifier != null) {
70        collected.add(identifier);
71      }
72    }
73  
74    /**
75     * Return the identifiers collected on this thread, in encounter order.
76     *
77     * @return the identifiers, empty if no rebuild is running
78     */
79    public static Set<String> collected() {
80      Set<String> collected = COLLECTED.get();
81      return collected == null ? Collections.emptySet() : Collections.unmodifiableSet(collected);
82    }
83  
84    /**
85     * Stop collecting on the current thread and release the collected identifiers.
86     */
87    public static void stop() {
88      COLLECTED.remove();
89    }
90  
91  }
