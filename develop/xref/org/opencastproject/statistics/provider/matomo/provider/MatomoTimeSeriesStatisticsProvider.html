22  package org.opencastproject.statistics.provider.matomo.provider;
23  
24  import org.opencastproject.statistics.api.DataResolution;
25  import org.opencastproject.statistics.api.ResourceType;
26  import org.opencastproject.statistics.api.TimeSeries;
27  import org.opencastproject.statistics.api.TimeSeriesProvider;
28  import org.opencastproject.statistics.provider.matomo.StatisticsProviderMatomoService;
29  
30  import java.time.Instant;
31  import java.time.ZoneId;
32  import java.util.Map;
33  import java.util.Set;
34  
35  
36  public class MatomoTimeSeriesStatisticsProvider extends MatomoStatisticsProvider implements TimeSeriesProvider {
37  
38  
39    private final MatomoProviderConfiguration.MatomoProviderSource source;
40    private final BatchMatomoRequest batchRequest;
41  
42    public MatomoTimeSeriesStatisticsProvider(
43        StatisticsProviderMatomoService service,
44        String id,
45        ResourceType resourceType,
46        String title,
47        String description,
48        Set<MatomoProviderConfiguration.MatomoProviderSource> sources
49    ) {
50      super(service, id, resourceType, title, description);
51      if (sources.isEmpty()) {
52        throw new IllegalArgumentException("At least one source must be provided");
53      }
54      if (sources.size() > 1) {
55        throw new IllegalArgumentException("Currently only one source is supported");
56      }
57      this.source = sources.iterator().next();
58  
59      BatchMatomoRequest request = service.getBatchRequest(resourceType.toString() + source.getMethod());
60      if (request == null) {
61        request = new BatchMatomoRequest(service, source.getMethod());
62        service.registerBatchRequest(resourceType.toString() + source.getMethod(), request);
63      }
64      request.addProvider(this, source.getAggregationVariable(), source.getAggregation());
65      this.batchRequest = request;
66    }
67  
68    @Override
69    public TimeSeries getValues(String resourceId, Instant from, Instant to, DataResolution resolution, ZoneId zoneId) {
70      if (!source.getResolutions().contains(resolution)) {
71        throw new IllegalArgumentException("Resolution " + resolution + " not supported by this provider");
72      }
73      final String matomoPeriod = dataResolutionToMatomoPeriod(resolution);
74  
75      // Execute batch request
76      Map<String, TimeSeries> results = batchRequest.executeRequest(
77          resourceId,
78          from,
79          to,
80          matomoPeriod,
81          source.getSiteId(),
82          source.getDimensionId(),
83          zoneId,
84          resolution
85      );
86  
87      // Return this provider's result
88      return results.get(getId());
89    }
90  
91    @Override
92    public Set<DataResolution> getDataResolutions() {
93      if (source == null) {
94        throw new IllegalStateException("Source configuration is not initialized");
95      }
96      return source.getResolutions();
97    }
98  
99  }
