74  /**
75   * Periodically checks on the tasks in the database.
76   * If there are scheduled tasks this attempts to run them.
77   * If there are running tasks this attempts to check if their status should be changed
78   */

146   @Activate
147   public void activate(ComponentContext cc) throws ConfigurationException {
148     logger.info("Activating LifeCycle Management Task Runner.");
149     this.defaultOrganization = new DefaultOrganization();
150     String systemAdminUserName = cc.getBundleContext().getProperty(SecurityUtil.PROPERTY_KEY_SYS_USER);
151     this.systemAdminUser = SecurityUtil.createSystemUser(systemAdminUserName, defaultOrganization);
152 
153     scheduledExecutor = Executors.newScheduledThreadPool(1);
154     scheduledExecutor.scheduleWithFixedDelay(new LifeCycleTaskRunner.Runner(), 10, 5,
155         TimeUnit.SECONDS);
156 
157     updated(cc.getProperties());
158   }
159 
160   @Modified
161   public void modified(ComponentContext cc) throws ConfigurationException {
162     updated(cc.getProperties());
163   }
164 
165   public void updated(Dictionary<String, ?> properties) throws ConfigurationException {
166     if (properties == null) {
167       return;
168     }
169     retractWorkflowId = StringUtils.defaultString((String) properties.get(RETRACT_WORKFLOW), DEFAULT_RETRACT_WORKFLOW);
170     logger.debug("Retract Workflow is '{}'", retractWorkflowId);
171   }

173   @Deactivate
174   public void deactivate() {
175     if (scheduledExecutor != null) {
176       try {
177         scheduledExecutor.shutdownNow();
178         if (!scheduledExecutor.isShutdown()) {
179           logger.info("Waiting for Dispatcher to terminate");
180           scheduledExecutor.awaitTermination(10, TimeUnit.SECONDS);
181         }
182       } catch (InterruptedException e) {
183         logger.error("Error shutting down the Dispatcher", e);
184       }
185     }
186   }

188   class Runner implements Runnable {
189     @Override
190     public void run() {
191       logger.debug("LifeCycleTaskRunner runs");
192 
193       List<Organization> orgs = organizationDirectoryService.getOrganizations();
194 
195       for (Organization org : orgs) {
196         SecurityUtil.runAs(securityService, org, systemAdminUser, () -> {
197 
198           // Get tasks that are ready to be run
199           List<LifeCycleTask> tasks;
200           try {
201             tasks = lifeCycleService.getLifeCycleTasksWithStatus(Status.SCHEDULED);
202           } catch (NullPointerException | LifeCycleServiceException e) {
203             logger.warn("Could not get lifecycle tasks in status 'SCHEDULED' from service: ", e);
204             return;
205           }
206 
207           // Get number of tasks that are running
208           int numberOfRunningTasks = 0;
209           try {
210             numberOfRunningTasks = lifeCycleService.getLifeCycleTasksWithStatus(Status.STARTED).size();
211           } catch (NullPointerException | LifeCycleServiceException e) {
212             logger.warn("Could not get lifecycle tasks in status 'STARTED' from service: ", e);
213             return;
214           }
215 
216           // For each task
217           for (LifeCycleTask task : tasks) {
218             // Prevent too many tasks from running at the same time to avoid system overload
219             if (numberOfRunningTasks > maxConcurrentTasks) {
220               logger.debug("The number of running tasks is too high at " + numberOfRunningTasks + ". Not starting more "
221                   + "tasks until it lowers below " + maxConcurrentTasks + ".");
222               break;
223             }
224             // Check action and do action related things
225             // TODO: Limit number of retries in case of error?
226             try {
227               logger.debug("SCHEDULED Task " + task.getId());
228               LifeCyclePolicy policy = lifeCycleService.getLifeCyclePolicyById(task.getLifeCyclePolicyId());
229               switch(policy.getAction()) {
230                 case START_WORKFLOW -> startWorkflow((LifeCycleTaskStartWorkflow)task, policy);
231                 case DELETE_EVENT -> deleteEvent(task, policy);
232                 default -> throw new NotImplementedException();
233               }
234               numberOfRunningTasks++;
235             } catch (NotFoundException e) {
236               logger.warn("Could not start action for task with id " + task.getId() + ". Lifecycle Policy with id "
237                   + task.getLifeCyclePolicyId() + " was not found.");
238             } catch (UnauthorizedException e) {
239               logger.warn("Could not start action for task with id " + task.getId() + ". User "
240                   + securityService.getUser().getUsername() + " was not authorized to access "
241                   + "lifecycle policy " + task.getLifeCyclePolicyId());
242             } catch (LifeCycleServiceException | RuntimeException e) {
243               logger.warn("Could not start action for task with id " + task.getId() + " based of policy "
244                   + task.getLifeCyclePolicyId());
245               logger.warn(e.toString());
246             }
247           }
248 
249           // Get tasks that are running
250           try {
251             tasks = lifeCycleService.getLifeCycleTasksWithStatus(Status.STARTED);
252           } catch (NullPointerException | LifeCycleServiceException e) {
253             logger.warn("Could not get lifecycle tasks in status 'STARTED' from service: ", e);
254             return;
255           }
256 
257           // For each task
258           for (LifeCycleTask task : tasks) {
259             // Check action and do action related things
260             try {
261               logger.debug("STARTED Task " + task.getId());
262               LifeCyclePolicy policy = lifeCycleService.getLifeCyclePolicyById(task.getLifeCyclePolicyId());
263               switch(policy.getAction()) {
264                 case START_WORKFLOW -> checkIfWorkflowDone((LifeCycleTaskStartWorkflow)task);
265                 default -> throw new NotImplementedException();
266               }
267             } catch (NotFoundException e) {
268               logger.warn("Could not check on running task with id " + task.getId() + ". Lifecycle Policy with id "
269                   + task.getLifeCyclePolicyId() + " was not found.");
270             } catch (UnauthorizedException e) {
271               logger.warn("Could not check on running task with id " + task.getId() + ". User "
272                   + securityService.getUser().getUsername() + " was not authorized to access "
273                   + "lifecycle policy " + task.getLifeCyclePolicyId());
274             } catch (LifeCycleServiceException e) {
275               logger.warn("Could not check on running task with id " + task.getId() + " based of policy "
276                   + task.getLifeCyclePolicyId());
277               logger.warn(e.toString());
278             }
279           }
280 
281         });
282       }
283     }

286   /**
287    * For a given task based on the START_WORKFLOW action, start a workflow
288    * @param task the life cycle task
289    * @param policy the life cycle policy
290    * @throws NotFoundException If the workflow definition, or the persisted task, could not be found
291    * @throws LifeCycleServiceException If something went wrong
292    * @throws UnauthorizedException If not authorized to start the workflow, or to persist the task
293    */
294   private void startWorkflow(LifeCycleTaskStartWorkflow task, LifeCyclePolicy policy)
295           throws NotFoundException, LifeCycleServiceException, UnauthorizedException {
296     String mediaPackageId = task.getTargetId();
297     String actionParameters = policy.getActionParameters();
298 
299     if (policy.getTargetType() != TargetType.EVENT) {
300       throw new IllegalArgumentException(
301           "The action START_WORKFLOW is only supported for the targetType EVENT. Given target type was: "
302               + policy.getAction()
303       );
304     }
305 
306     // Parse Parameters
307     StartWorkflowParameters actionParametersParsed = gson.fromJson(actionParameters, StartWorkflowParameters.class);
308     if (actionParametersParsed.getWorkflowId() == null) {
309       task.setStatus(Status.FAILED);
310       lifeCycleService.updateLifeCycleTask(task);
311       throw new IllegalArgumentException(
312           "No workflowId found. Should have been at least the workflowId."
313       );
314     }
315     String workflowId = actionParametersParsed.getWorkflowId();
316     Map<String, String> workflowParameters;
317     if (actionParametersParsed.getWorkflowParameters() == null) {
318       workflowParameters = WorkflowPropertiesUtil.getLatestWorkflowProperties(assetManager, mediaPackageId);
319     } else {
320       workflowParameters = actionParametersParsed.getWorkflowParameters();
321     }
322 
323     // Get mediapackage
324     Optional<MediaPackage> optMediaPackage = assetManager.getMediaPackage(mediaPackageId);
325     if (optMediaPackage.isEmpty()) {
326       throw new IllegalArgumentException(
327           "Mediapackage for given id " + mediaPackageId  + " does not exist."
328       );
329     }
330     MediaPackage mediaPackage = optMediaPackage.get();
331 
332     try {
333       // If there is currently an active workflow on the mediapackage, postpone this lifecycle task
334       Optional<WorkflowInstance> runningWfi = workflowService.getRunningWorkflowInstanceByMediaPackage(
335           mediaPackageId, Permissions.Action.READ.toString());
336       if (runningWfi.isPresent()) {
337         logger.info("Workflow running on " + mediaPackageId + ". Postponing execution of task " + task.getId());
338         return;
339       }
340 
341       // Start workflow
342       WorkflowInstance wfi = workflowService.start(
343           workflowService.getWorkflowDefinitionById(workflowId),
344           mediaPackage,
345           workflowParameters
346       );
347 
348       // Set task started when workflow is started
349       logger.info("Started task " + task.getId() + ". Starting " + workflowId + " on " + mediaPackageId);
350       task.setStatus(Status.STARTED);
351       task.setWorkflowInstanceId(wfi.getId());
352       lifeCycleService.updateLifeCycleTask(task);
353     } catch (WorkflowDatabaseException | WorkflowParsingException e) {
354       task.setStatus(Status.FAILED);
355       lifeCycleService.updateLifeCycleTask(task);
356       throw new LifeCycleServiceException(
357           "Unable to load workflow" + workflowId + " for mediapackage " + mediaPackageId + ". An error occurred.",
358           e);
359     } catch (NotFoundException e) {
360       task.setStatus(Status.FAILED);
361       lifeCycleService.updateLifeCycleTask(task);
362       throw new LifeCycleServiceException(
363           "Unable to load workflow" + workflowId + " for mediapackage " + mediaPackageId
364               + ". Workflow could not be found.",
365           e);
366     } catch (UnauthorizedException e) {
367       task.setStatus(Status.FAILED);
368       lifeCycleService.updateLifeCycleTask(task);
369       throw new LifeCycleServiceException(
370           "Not authorized to start workflow with id " + workflowId + " for mediapackage " + mediaPackageId,
371           e);
372     } catch (WorkflowException e) {
373       task.setStatus(Status.FAILED);
374       lifeCycleService.updateLifeCycleTask(task);
375       throw new LifeCycleServiceException(
376           "Unable to check if there is currently a workflow running on mediapackage " + mediaPackageId,
377           e);
378     }
379   }

381   /**
382    * For a given task that started a workflow, update the task status based on the workflow status
383    * @param task the life cycle task
384    */
385   private void checkIfWorkflowDone(LifeCycleTaskStartWorkflow task) {
386     try {
387       WorkflowInstance instance = workflowService.getWorkflowById(task.getWorkflowInstanceId());
388       WorkflowInstance.WorkflowState wfState = instance.getState();
389 
390       if (wfState == WorkflowInstance.WorkflowState.SUCCEEDED) {
391         task.setStatus(Status.FINISHED);
392         lifeCycleService.updateLifeCycleTask(task);
393       }
394       if (wfState == WorkflowInstance.WorkflowState.FAILED) {
395         task.setStatus(Status.FAILED);
396         lifeCycleService.updateLifeCycleTask(task);
397       }
398     } catch (WorkflowDatabaseException e) {
399       throw new RuntimeException(e);
400     } catch (NotFoundException e) {
401       throw new RuntimeException(e);
402     } catch (UnauthorizedException e) {
403       throw new RuntimeException(e);
404     } catch (LifeCycleServiceException e) {
405       throw new RuntimeException(e);
406     }
407   }

409   /**
410    */
411   private void deleteEvent(LifeCycleTask task, LifeCyclePolicy policy) throws LifeCycleServiceException {
412     String mediaPackageId = task.getTargetId();
413 
414     if (policy.getTargetType() != TargetType.EVENT) {
415       throw new IllegalArgumentException(
416           "The action DELETE_EVENT is only supported for the targetType EVENT. Given target type was: "
417               + policy.getAction()
418       );
419     }
420 
421     final Optional<Event> event;
422     try {
423       event = indexService.getEvent(mediaPackageId, opensearchIndex);
424 
425       if (event.isEmpty()) {
426         logger.debug("Event for given id " + mediaPackageId + " does not exist. No deletion necessary.");
427         return;
428       }
429 
430       IndexService.EventRemovalResult result = indexService.removeEvent(event.get(), retractWorkflowId);
431       if (result != IndexService.EventRemovalResult.GENERAL_FAILURE) {
432         task.setStatus(Status.FINISHED);
433       } else {
434         task.setStatus(Status.FAILED);
435       }
436       lifeCycleService.updateLifeCycleTask(task);
437     } catch (SearchIndexException | WorkflowDatabaseException e) {
438       task.setStatus(Status.FAILED);
439       throw new LifeCycleServiceException(
440           "Unable to find event for mediapackage " + mediaPackageId + ". An error occured",
441           e);
442     } catch (UnauthorizedException e) {
443       task.setStatus(Status.FAILED);
444       throw new LifeCycleServiceException(
445           "Not authorized to remove event with id " + mediaPackageId,
446           e);
447     } catch (NotFoundException e) {
448       task.setStatus(Status.FAILED);
449       throw new LifeCycleServiceException(
450           "Unable to find event " + mediaPackageId,
451           e);
452     }
453   }
