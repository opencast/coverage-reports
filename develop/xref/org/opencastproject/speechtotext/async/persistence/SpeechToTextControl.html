45  @Entity(name = "SpeechToTextControl")
46  //@formatter:off
47  @Table(name = "oc_speech_to_text_control", indexes = {
48          @Index(name = "IX_oc_stt_status", columnList = ("status")),
49          @Index(name = "IX_oc_stt_workflow_id", columnList = ("workflow_id")),
50          @Index(name = "IX_oc_stt_job", columnList = ("job"))
51  })
52  @NamedQueries({
53          @NamedQuery(name = "SpeechToTextControl.findByWorkflowId",
54              query = "SELECT stt FROM SpeechToTextControl stt "
55                  + "WHERE stt.workflowId = :workflowId"),
56          @NamedQuery(name = "SpeechToTextControl.findDistinctWorkflowIdByStatus",
57              query = "SELECT DISTINCT stt.workflowId FROM SpeechToTextControl stt "
58                  + "WHERE stt.status IN :status"),
59          @NamedQuery(name = "SpeechToTextControl.findDistinctWorkflowIdByMediaPackage",
60          query = "SELECT DISTINCT stt.workflowId FROM SpeechToTextControl stt "
61              + "WHERE stt.mediaPackageId = :mpId"),
62          @NamedQuery(name = "SpeechToTextControl.findByJob",
63              query = "SELECT stt FROM SpeechToTextControl stt WHERE stt.job = :job"),
64          @NamedQuery(name = "SpeechToTextControl.findByStatus",
65              query = "SELECT stt FROM SpeechToTextControl stt WHERE stt.status IN :status"),
66          @NamedQuery(name = "SpeechToTextControl.updateStatusByJob",
67              query = "UPDATE SpeechToTextControl stt SET stt.status = :status "
68                  + "WHERE stt.job IN :jobs"),
69          @NamedQuery(name = "SpeechToTextControl.deleteByWorkflow",
70          query = "DELETE FROM SpeechToTextControl stt WHERE stt.workflowId = :workflowId"),
71          @NamedQuery(name = "SpeechToTextControl.updateStatusByStatusAndDate",
72              query = "UPDATE SpeechToTextControl stt SET stt.status = :newStatus "
73                  + "WHERE stt.status IN :oldStatus and stt.dateCreated < :date")})
74  // @formatter:on

76  /**
77   * Controls the status of speech to text jobs; used by the quartz job that starts a workflow to asynchronously attach
78   * the captions to a media package.
79   */
80  public class SpeechToTextControl {
81    @Id
82    @GeneratedValue(strategy = GenerationType.AUTO)
83    @Column(name = "id", length = 128)
84    @XmlAttribute
85    private long id;
86  
87    @Column(name = "mediapackage_id", nullable = false, length = 128)
88    private String mediaPackageId;
89  
90    /*
91     * Workflow that started STT. Workflow id is used to group related speech to text jobs; it's possible that more than
92     * one workflow ran on the same media package, generating more than one stt job groups.
93     */
94    @Column(name = "workflow_id", nullable = false)
95    private long workflowId;
96  
97    @OneToOne(targetEntity = JpaJob.class, optional = false, cascade = { CascadeType.REMOVE })
98    @JoinColumn(name = "job", referencedColumnName = "id")
99    private JpaJob job = null;
100 
101   @Column(name = "date_created", nullable = false)
102   @Temporal(TemporalType.TIMESTAMP)
103   private Date dateCreated;
104 
105   @Column(name = "date_completed", nullable = true)
106   @Temporal(TemporalType.TIMESTAMP)
107   private Date dateCompleted;
108 
109   @Column(name = "status", nullable = false, length = 128)
110   @Enumerated(EnumType.STRING)
111   private SpeechToTextControl.Status status;
112 
113   public enum Status {
114     InProgress, // Transcription running or stt job queued
115     TranscriptionDone, // Transcription done (ok), captions not attached yet
116     TranscriptionError, // Transcription done (error), no action taken yet
117     WorkflowInProgress, // Workflow to attach transcripts created by the quartz task
118     Canceled, // An error occurred e.g. subtitle generation error occurred and generation rescheduled,
119               // subtitle generation error occurred and no retry, subtitle generation never finished,
120               // workflow to attach subtitles could not be started after allotted interval
121     Done // Done, transcription attached to media package
122   }
123 
124   public SpeechToTextControl() {
125   }
126 
127   /**
128    * All fields constructor.
129    */
130   public SpeechToTextControl(long id, String mediaPackageId, long workflowId, JpaJob job, Date dateStarted,
131           Date dateCompleted, Status status) {
132     super();
133     this.id = id;
134     this.mediaPackageId = mediaPackageId;
135     this.workflowId = workflowId;
136     this.job = job;
137     this.dateCreated = dateStarted;
138     this.dateCompleted = dateCompleted;
139     this.status = status;
140   }
141 
142   /**
143    * Creates a stt control.
144    *
145    * @param mediaPackageId
146    *          the media package id
147    * @param workflowId
148    *          workflow id that created the stt jobs
149    * @param job
150    *          associated stt job
151    * @param dateCreated
152    *          date created
153    * @param status
154    *          status
155    */
156   public SpeechToTextControl(String mediaPackageId, long workflowId, JpaJob job, Date dateCreated,
157           SpeechToTextControl.Status status) {
158     super();
159     this.mediaPackageId = mediaPackageId;
160     this.workflowId = workflowId;
161     this.job = job;
162     this.dateCreated = dateCreated;
163     this.status = status;
164   }
165 
166   public long getId() {
167     return this.id;
168   }
169 
170   public Date getDateCreated() {
171     return this.dateCreated;
172   }
173 
174   public Date getDateCompleted() {
175     return this.dateCompleted;
176   }
177 
178   public String getMediaPackageId() {
179     return mediaPackageId;
180   }
181 
182   public Status getStatus() {
183     return this.status;
184   }
185 
186   public JpaJob getJob() {
187     return this.job;
188   }
189 
190 }
