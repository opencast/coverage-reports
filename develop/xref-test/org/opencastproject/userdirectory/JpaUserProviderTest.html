78    @Test
79    public void testAddAndGetUser() throws Exception {
80      JpaUser user = createUserWithRoles(org1, "user1", "ROLE_ASTRO_101_SPRING_2011_STUDENT");
81      provider.addUser(user);
82  
83      User loadUser = provider.loadUser("user1");
84      assertNotNull(loadUser);
85  
86      assertEquals(user.getUsername(), loadUser.getUsername());
87      assertTrue(passwordEncoder.matches(user.getPassword(), loadUser.getPassword()));
88      assertEquals(user.getOrganization(), loadUser.getOrganization());
89      assertEquals(user.getRoles(), loadUser.getRoles());
90  
91      assertNull("Loading 'does not exist' should return null", provider.loadUser("does not exist"));
92      assertNull("Loading 'does not exist' should return null", provider.loadUser("user1", org2.getId()));
93  
94      loadUser = provider.loadUser("user1", org1.getId());
95      assertNotNull(loadUser);
96  
97      assertEquals(user.getUsername(), loadUser.getUsername());
98      assertTrue(passwordEncoder.matches(user.getPassword(), loadUser.getPassword()));
99      assertEquals(user.getOrganization(), loadUser.getOrganization());
100     assertEquals(user.getRoles(), loadUser.getRoles());
101   }

162   @Test
163   public void testDeleteUser() throws Exception {
164     JpaUser user1 = createUserWithRoles(org1, "user1", "ROLE_ASTRO_101_SPRING_2011_STUDENT");
165     JpaUser user2 = createUserWithRoles(org1, "user2", "ROLE_ASTRO_101_SPRING_2011_STUDENT");
166     JpaUser user3 = createUserWithRoles(org1, "user3", "ROLE_ASTRO_101_SPRING_2011_STUDENT");
167     JpaUser user4 = createUserWithRoles(org1, "user4", "ROLE_ASTRO_101_SPRING_2011_STUDENT");
168     provider.addUser(user1);
169     provider.addUser(user2);
170     provider.addUser(user3);
171     provider.addUser(user4);
172 
173     User loadUser = provider.loadUser("user1");
174 
175     assertNotNull(loadUser);
176 
177     provider.deleteUser("user1", user1.getOrganization().getId());
178     provider.deleteUser("user2", user1.getOrganization().getId());
179     provider.deleteUser("user3", user1.getOrganization().getId());
180 
181     assertNull(provider.loadUser("user1", org1.getId()));
182     assertNull(provider.loadUser("user2", org1.getId()));
183     assertNull(provider.loadUser("user3", org1.getId()));
184     assertNotNull(provider.loadUser("user4", org1.getId()));
185 
186     try {
187       provider.deleteUser("user1", user1.getOrganization().getId());
188       fail("Should throw a NotFoundException");
189     } catch (NotFoundException e) {
190       assertTrue("User not found.", true);
191     }
192   }

213     authorities.add(new JpaRole("ROLE_ASTRO_101_SPRING_2011_STUDENT", org1));

234

246     JpaUser adminUser = createUserWithRoles(org1, "admin", SecurityConstants.GLOBAL_ADMIN_ROLE);
247     JpaUser user = createUserWithRoles(org1, "user", "ROLE_USER");
248 
249     provider.addUser(adminUser);

251     provider.setSecurityService(mockSecurityServiceWithUser(user));
252 
253     // try to add ROLE_USER
254     Set<JpaRole> updatedRoles = Set.of(
255             new JpaRole("ROLE_USER", org1),
256             new JpaRole(SecurityConstants.GLOBAL_ADMIN_ROLE, org1));
257 
258     try {
259       provider.updateUser(new JpaUser(adminUser.getUsername(), adminUser.getPassword(), org1,
260               adminUser.getName(), true, updatedRoles));
261       fail("The current user may not edit an admin user");
262     } catch (UnauthorizedException e) {
263       // pass
264     }
265 
266     // try to remove ROLE_ADMIN
267     updatedRoles = Set.of(new JpaRole("ROLE_USER", org1));
268     try {
269       provider.updateUser(new JpaUser(adminUser.getUsername(), adminUser.getPassword(), org1,
270               adminUser.getName(), true, updatedRoles));
271       fail("The current user may not remove the admin role on other user");
272     } catch (UnauthorizedException e) {
273       // pass
274     }

331   @Test
332   public void testRolesForUser() {
333     JpaRole astroRole = new JpaRole("ROLE_ASTRO_105_SPRING_2013_STUDENT", org1, "Astro role");
334 
335     provider.addRole(astroRole);
336 
337     JpaUser userOne = createUserWithRoles(org1, "user1", "ROLE_ONE", "ROLE_TWO");
338     try {
339       provider.addUser(userOne);
340     } catch (UnauthorizedException e) {
341       fail("User should be created");
342     }
343 
344     // Provider not authoritative for these roles
345     assertEquals(
346         "There should be zero roles",
347         0,
348         IteratorUtils.toList(provider.findRoles("%", Role.Target.ALL, 0, 0)).size()
349     );
350 
351     List<String> rolesForUser = provider.getRolesForUser("user1").stream()
352             .map(Role::getName)
353             .sorted()
354             .collect(Collectors.toList());
355     assertEquals("There should be two roles", 2, rolesForUser.size());
356     assertEquals("ROLE_ONE", rolesForUser.get(0));
357     assertEquals("ROLE_TWO", rolesForUser.get(1));
358   }

395   private static SecurityService mockSecurityServiceWithUser(User currentUser) {
396     // Set the security sevice
397     SecurityService securityService = EasyMock.createNiceMock(SecurityService.class);
398     EasyMock.expect(securityService.getOrganization()).andReturn(currentUser.getOrganization()).anyTimes();
399     EasyMock.expect(securityService.getUser()).andReturn(currentUser).anyTimes();
400     EasyMock.replay(securityService);
401     return securityService;
402   }

404   private static JpaUser createUserWithRoles(JpaOrganization org, String username, String... roles) {
405     Set<JpaRole> userRoles = new HashSet<>();
406     for (String adminRole : roles) {
407       userRoles.add(new JpaRole(adminRole, org));
408     }
409     return new JpaUser(username, "pass1", org, "opencast", true, userRoles);
410   }
