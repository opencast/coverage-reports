21  package org.opencastproject.speechtotext.async.impl;
22  
23  import static org.opencastproject.db.DBTestEnv.getDbSessionFactory;
24  import static org.opencastproject.db.DBTestEnv.newEntityManagerFactory;
25  import static org.opencastproject.speechtotext.async.persistence.SpeechToTextTestUtil.createJob;
26  
27  import org.opencastproject.db.DBSession;
28  import org.opencastproject.db.DBSessionFactory;
29  import org.opencastproject.job.api.Job;
30  import org.opencastproject.job.jpa.JpaJob;
31  import org.opencastproject.speechtotext.async.persistence.SpeechToTextControl;
32  import org.opencastproject.speechtotext.async.persistence.SpeechToTextDatabase;
33  
34  import org.junit.Assert;
35  import org.junit.Before;
36  import org.junit.Test;
37  
38  import java.util.Date;
39  import java.util.List;
40  
41  import javax.persistence.EntityManagerFactory;
42  
43  public class SpeechToTextAsyncTrackerImplTest {
44    private static final long JOB_ID_1 = 1L;
45    private static final long JOB_ID_2 = 2L;
46    private static final long WF_ID = 1L;
47    private static final String MP_ID = "media_package_id";
48  
49    private DBSession db;
50  
51    private SpeechToTextDatabase database;
52    private SpeechToTextAsyncTrackerImpl tracker;
53  
54    @Before
55    public void setUp() {
56      EntityManagerFactory emf = newEntityManagerFactory(SpeechToTextDatabase.PERSISTENCE_UNIT);
57      DBSessionFactory dbSessionFactory = getDbSessionFactory();
58      db = dbSessionFactory.createSession(emf);
59  
60      database = new SpeechToTextDatabase();
61      database.setEntityManagerFactory(emf);
62      database.setDBSessionFactory(dbSessionFactory);
63      database.activate(null);
64  
65      tracker = new SpeechToTextAsyncTrackerImpl();
66      tracker.setSpeechToTextDatabase(database);
67    }
68  
69    @Test
70    public void testTrack() throws Exception {
71      JpaJob job1 = createJob(db, new Date(), Job.Status.INSTANTIATED, JOB_ID_1);
72      tracker.track(job1.toJob(), MP_ID, WF_ID);
73      JpaJob job2 = createJob(db, new Date(), Job.Status.INSTANTIATED, JOB_ID_2);
74      tracker.track(job2.toJob(), MP_ID, WF_ID);
75  
76      List<SpeechToTextControl> stt = database.findByWorkflowId(WF_ID);
77      Assert.assertEquals(2, stt.size());
78      Assert.assertEquals(MP_ID, stt.get(0).getMediaPackageId());
79      Assert.assertEquals(MP_ID, stt.get(1).getMediaPackageId());
80      Assert.assertEquals(SpeechToTextControl.Status.InProgress, stt.get(0).getStatus());
81      Assert.assertEquals(SpeechToTextControl.Status.InProgress, stt.get(1).getStatus());
82    }
83  
84    @Test
85    public void testUntrack() throws Exception {
86      JpaJob job1 = createJob(db, new Date(), Job.Status.INSTANTIATED, JOB_ID_1);
87      tracker.track(job1.toJob(), MP_ID, WF_ID);
88  
89      tracker.untrack(job1.toJob());
90  
91      SpeechToTextControl stt = database.findByJob(job1);
92      Assert.assertNotNull(stt);
93      Assert.assertEquals(MP_ID, stt.getMediaPackageId());
94      Assert.assertEquals(SpeechToTextControl.Status.Done, stt.getStatus());
95    }
96  
97  }
