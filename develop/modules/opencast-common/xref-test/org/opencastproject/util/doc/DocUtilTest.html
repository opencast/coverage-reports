37  /**
38   * Tests that the FreeMarker configuration used by {@link DocUtil} is safe for templates whose content or data
39   * is influenced by user-controlled values (e.g. operation configuration values and workflow properties). The
40   * engine must never allow template content to instantiate classes, access the Java API or load other
41   * templates, and template errors must not leak the template or the data into the thrown exception (whose
42   * stack trace is stored in incident details).
43   */
44  public class DocUtilTest {
45  
46    /**
47     * @param entries
48     *          key/value pairs
49     * @return a data model
50     */
51    private static Map<String, Object> data(String... entries) {
52      Map<String, Object> data = new HashMap<>();
53      for (int i = 0; i < entries.length; i += 2) {
54        data.put(entries[i], entries[i + 1]);
55      }
56      return data;
57    }
58  
59    @Test
60    public void interpolatesData() {
61      assertEquals("Hello World!", DocUtil.processTextTemplate("test", "Hello ${name}!", data("name", "World")));
62    }
63  
64    @Test
65    public void emptyDataModelDoesNotReturnTheTemplateUnprocessed() {
66      // Static templates are rendered with an empty (or null) data model, and templates referencing
67      // missing variables fail instead of passing through unrendered
68      assertEquals("static text", DocUtil.processTextTemplate("test", "static text", new HashMap<>()));
69      assertEquals("static text", DocUtil.processTextTemplate("test", "static text", null));
70      try {
71        DocUtil.processTextTemplate("test", "${missing}", new HashMap<>());
72        fail("Expected a RuntimeException for a template that references a missing variable");
73      } catch (RuntimeException e) {
74        // expected
75      }
76    }
77  
78    @Test
79    public void supportsDefaultValues() {
80      assertEquals("fallback", DocUtil.processTextTemplate("test", "${missing!'fallback'}", data("name", "value")));
81    }

83    @Test
84    public void supportsConditionals() {
85      assertEquals("yes", DocUtil.processTextTemplate("test", "<#if flag == 'on'>yes<#else>no</#if>",
86              data("flag", "on")));
87    }

89    @Test
90    public void cannotInstantiateClasses() throws Exception {
91      Path marker = Files.createTempFile("docutil-ssti", null);
92      Files.delete(marker);
93      try {
94        // The command is only escaped for the FreeMarker string literal: FreeMarker's Execute uses
95        // Runtime.exec(String), which splits the command on whitespace without a shell, so shell
96        // quoting would end up in the path as literal characters
97        Assume.assumeFalse("The marker path must not contain whitespace for the payload to work",
98                marker.toAbsolutePath().toString().matches(".*\\s.*"));
99        String command = "touch " + marker.toAbsolutePath();
100       String payload = "${\"freemarker.template.utility.Execute\"?new()(" + freeMarkerString(command) + ")}";
101       try {
102         DocUtil.processTextTemplate("test", payload, data("name", "value"));
103         fail("Expected a RuntimeException: class instantiation must be blocked");
104       } catch (RuntimeException e) {
105         // Assert on the full stack trace including the cause chain: incident details store it,
106         // so the template must not leak through a cause either
107         String stackTrace = ExceptionUtils.getStackTrace(e);
108         assertFalse("The template source must not leak into the exception or its cause chain, got: " + stackTrace,
109                 stackTrace.contains(command));
110         assertFalse("The template must not be able to instantiate classes (e.g. to execute commands)",
111                 Files.exists(marker));
112       }
113     } finally {
114       Files.deleteIfExists(marker);
115     }
116   }

118   @Test
119   public void cannotAccessTheApi() {
120     // With the API built-in disabled, ?api fails while the template is processed; if it were enabled,
121     // the template would render the object's string representation instead
122     try {
123       DocUtil.processTextTemplate("test", "${name?api}", data("name", "value"));
124       fail("Expected a RuntimeException: the ?api built-in must be disabled");
125     } catch (RuntimeException e) {
126       // expected
127     }
128   }
129 
130   @Test
131   public void cannotIncludeTemplates() {
132     try {
133       DocUtil.processTextTemplate("test", "<#include \"/etc/passwd\">", data("name", "value"));
134       fail("Expected a RuntimeException: #include must fail without a template loader");
135     } catch (RuntimeException e) {
136       // The include fails before any file is read; make sure that nothing of the file leaked anyway
137       String stackTrace = ExceptionUtils.getStackTrace(e);
138       assertFalse("The included file's content must not leak into the exception, got: " + stackTrace,
139               stackTrace.contains("root:"));
140     }
141   }
142 
143   @Test
144   public void processErrorDoesNotLeakTemplateOrData() {
145     try {
146       DocUtil.processTextTemplate("test",
147               "${\"freemarker.template.utility.Execute\"?new()(\"id\")}",
148               data("name", "value", "secret", "super-secret-value"));
149       fail("Expected a RuntimeException");
150     } catch (RuntimeException e) {
151       // Assert on the full stack trace including the cause chain: incident details store it,
152       // so the template and the data must not leak through a cause either
153       String stackTrace = ExceptionUtils.getStackTrace(e);
154       assertFalse("The data model must not leak into the exception or its cause chain",
155               stackTrace.contains("super-secret-value"));
156       assertFalse("The template source must not leak into the exception or its cause chain",
157               stackTrace.contains("freemarker.template.utility.Execute"));
158     }
159   }
160 
161   @Test
162   public void parseErrorDoesNotLeakTemplate() {
163     String payload = "${top-secret-payload-12345";
164     try {
165       DocUtil.processTextTemplate("test", payload, data("name", "value"));
166       fail("Expected a RuntimeException for an invalid template");
167     } catch (RuntimeException e) {
168       // Assert on the full stack trace including the cause chain: incident details store it,
169       // so the template must not leak through a cause either
170       String stackTrace = ExceptionUtils.getStackTrace(e);
171       assertFalse("The template source must not leak into the exception or its cause chain, got: " + stackTrace,
172               stackTrace.contains("top-secret-payload-12345"));
173     }
174   }

176   /**
177    * @param value
178    *          a string
179    * @return the value escaped for use inside a double-quoted FreeMarker string literal
180    */
181   private static String freeMarkerString(String value) {
182     return "\"" + value.replace("\\", "\\\\").replace("\"", "\\\"") + "\"";
183   }
