22  package org.opencastproject.util.doc;
23  
24  import org.junit.Assert;
25  import org.junit.Test;
26  
27  import java.util.HashMap;
28  import java.util.LinkedHashMap;
29  import java.util.Map;
30  
31  public class OrganizationTemplateDataTest {
32  
33    @Test
34    public void nullPropertiesAreIgnored() {
35      Map<String, Object> dataModel = new HashMap<>();
36      OrganizationTemplateData.addOrganizationProperties(dataModel, null);
37      Assert.assertTrue(dataModel.isEmpty());
38    }
39  
40    @Test
41    public void undottedPropertiesAreAvailableInTheFlattenedForm() {
42      Map<String, Object> dataModel = new HashMap<>();
43      Map<String, String> properties = new LinkedHashMap<>();
44      properties.put("transcriber", "azure-1");
45  
46      OrganizationTemplateData.addOrganizationProperties(dataModel, properties);
47  
48      Assert.assertEquals("azure-1", dataModel.get("org_transcriber"));
49      Assert.assertEquals(1, dataModel.size());
50    }
51  
52    @Test
53    public void dottedPropertiesAreAvailableInTheDocumentedAndTheFlattenedForm() {
54      Map<String, Object> dataModel = new HashMap<>();
55      Map<String, String> properties = new LinkedHashMap<>();
56      properties.put("org.opencastproject.admin.ui.url", "http://admin.example.com");
57  
58      OrganizationTemplateData.addOrganizationProperties(dataModel, properties);
59  
60      // The flattened form
61      Assert.assertEquals("http://admin.example.com", dataModel.get("org_org_opencastproject_admin_ui_url"));
62      // The dotted form, as documented in the workflow configuration guide: ${org_org.opencastproject.admin.ui.url}
63      Map org = (Map) dataModel.get("org_org");
64      Assert.assertNotNull(org);
65      Map opencastproject = (Map) org.get("opencastproject");
66      Assert.assertNotNull(opencastproject);
67      Map admin = (Map) opencastproject.get("admin");
68      Assert.assertNotNull(admin);
69      Map ui = (Map) admin.get("ui");
70      Assert.assertNotNull(ui);
71      Assert.assertEquals("http://admin.example.com", ui.get("url"));
72    }
73  
74    @Test
75    public void propertiesSharingAPrefixShareTheNestedStructure() {
76      Map<String, Object> dataModel = new HashMap<>();
77      Map<String, String> properties = new LinkedHashMap<>();
78      properties.put("org.opencastproject.admin.ui.url", "http://admin.example.com");
79      properties.put("org.opencastproject.engage.ui.url", "http://engage.example.com");
80  
81      OrganizationTemplateData.addOrganizationProperties(dataModel, properties);
82  
83      Map org = (Map) dataModel.get("org_org");
84      Map opencastproject = (Map) org.get("opencastproject");
85      Map adminUi = (Map) ((Map) opencastproject.get("admin")).get("ui");
86      Assert.assertEquals("http://admin.example.com", adminUi.get("url"));
87      Map engageUi = (Map) ((Map) opencastproject.get("engage")).get("ui");
88      Assert.assertEquals("http://engage.example.com", engageUi.get("url"));
89    }
90  
91    @Test
92    public void aPropertyThatIsADotPrefixOfAnotherKeepsOnlyItsFlattenedForm() {
93      Map<String, Object> dataModel = new HashMap<>();
94      Map<String, String> properties = new LinkedHashMap<>();
95      properties.put("a.b", "short-value");
96      properties.put("a.b.c", "long-value");
97  
98      OrganizationTemplateData.addOrganizationProperties(dataModel, properties);
99  
100     // Both properties keep their flattened form
101     Assert.assertEquals("short-value", dataModel.get("org_a_b"));
102     Assert.assertEquals("long-value", dataModel.get("org_a_b_c"));
103     // The nested structure is created by the longer key; the shorter key's dotted form is a map, not its value
104     Map orgA = (Map) dataModel.get("org_a");
105     Assert.assertTrue(orgA.get("b") instanceof Map);
106     Assert.assertEquals("long-value", ((Map) orgA.get("b")).get("c"));
107   }
108 
109   @Test
110   public void aDottedPropertyExtendingAnUndottedOneKeepsOnlyItsFlattenedForm() {
111     Map<String, Object> dataModel = new HashMap<>();
112     Map<String, String> properties = new LinkedHashMap<>();
113     properties.put("player", "/play/");
114     properties.put("player.mobile", "/m/");
115 
116     OrganizationTemplateData.addOrganizationProperties(dataModel, properties);
117 
118     // Both properties keep their flattened form
119     Assert.assertEquals("/play/", dataModel.get("org_player"));
120     Assert.assertEquals("/m/", dataModel.get("org_player_mobile"));
121     // The undotted property occupies the root with its value, so the longer key has no dotted form
122     Assert.assertEquals(2, dataModel.size());
123   }
124 }
