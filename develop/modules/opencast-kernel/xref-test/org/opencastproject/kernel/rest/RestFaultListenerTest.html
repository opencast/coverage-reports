22  package org.opencastproject.kernel.rest;
23  
24  import static org.junit.Assert.assertEquals;
25  import static org.junit.Assert.assertFalse;
26  import static org.junit.Assert.assertNull;
27  import static org.junit.Assert.assertTrue;
28  
29  import org.apache.cxf.interceptor.Fault;
30  import org.apache.cxf.message.Exchange;
31  import org.apache.cxf.message.ExchangeImpl;
32  import org.apache.cxf.message.Message;
33  import org.apache.cxf.message.MessageImpl;
34  import org.apache.cxf.transport.http.AbstractHTTPDestination;
35  import org.easymock.EasyMock;
36  import org.junit.Before;
37  import org.junit.Test;
38  
39  import java.io.IOException;
40  import java.nio.channels.ClosedChannelException;
41  
42  import javax.servlet.http.HttpServletRequest;
43  import javax.ws.rs.BadRequestException;
44  import javax.ws.rs.InternalServerErrorException;
45  
46  public class RestFaultListenerTest {
47  
48    private final RestFaultListener listener = new RestFaultListener();
49    private Exchange exchange;
50    private Message outMessage;
51  
52    @Before
53    public void setUp() {
54      HttpServletRequest request = EasyMock.createMock(HttpServletRequest.class);
55      EasyMock.expect(request.getMethod()).andStubReturn("GET");
56      EasyMock.expect(request.getRequestURI()).andStubReturn("/admin-ng/event/events.json");
57      EasyMock.replay(request);
58  
59      exchange = new ExchangeImpl();
60      Message inMessage = new MessageImpl();
61      inMessage.put(AbstractHTTPDestination.HTTP_REQUEST, request);
62      inMessage.setExchange(exchange);
63      exchange.setInMessage(inMessage);
64  
65      outMessage = new MessageImpl();
66      outMessage.setExchange(exchange);
67      exchange.setOutMessage(outMessage);
68    }
69  
70    @Test
71    public void testDescribeRequest() {
72      assertEquals("GET /admin-ng/event/events.json", RestFaultListener.describeRequest(outMessage));
73      assertEquals("<unknown request>", RestFaultListener.describeRequest(new MessageImpl()));
74    }
75  
76    @Test
77    public void testIsClientDisconnect() {
78      assertTrue(RestFaultListener.isClientDisconnect(new Fault(new IOException(new ClosedChannelException()))));
79      assertFalse(RestFaultListener.isClientDisconnect(new Fault(new IOException("Disk full"))));
80      assertFalse(RestFaultListener.isClientDisconnect(new Fault(new NullPointerException())));
81    }
82  
83    @Test
84    public void testClientDisconnectSuppressesFollowUpFaults() {
85      assertFalse(listener.faultOccurred(new Fault(new ClosedChannelException()), "service", outMessage));
86      assertEquals(Boolean.TRUE, exchange.get(RestFaultListener.CLIENT_DISCONNECTED));
87      // CXF's attempt to write the error response fails as well
88      assertFalse(listener.faultOccurred(new Fault(new IOException("Closed")), "service", outMessage));
89    }
90  
91    @Test
92    public void testWebApplicationExceptionIsHandled() {
93      assertFalse(listener.faultOccurred(new BadRequestException(), "service", outMessage));
94      assertNull(exchange.get(RestFaultListener.CLIENT_DISCONNECTED));
95    }
96  
97    @Test
98    public void testFailedWriteOfMappedExceptionIsClientDisconnect() {
99      // JAXB providers wrap write failures in a WebApplicationException
100     Exception exception = new InternalServerErrorException(new IOException(new ClosedChannelException()));
101     assertFalse(listener.faultOccurred(exception, "service", outMessage));
102     assertEquals(Boolean.TRUE, exchange.get(RestFaultListener.CLIENT_DISCONNECTED));
103   }
104 
105   @Test
106   public void testOtherFaultsAreHandled() {
107     assertFalse(listener.faultOccurred(new Fault(new NullPointerException()), "service", outMessage));
108     assertNull(exchange.get(RestFaultListener.CLIENT_DISCONNECTED));
109   }
110 }
