21  package org.opencastproject.urlsigning.utils;

23  import org.opencastproject.urlsigning.common.Policy;
24  
25  import com.google.gson.Gson;
26  import com.google.gson.GsonBuilder;
27  import com.google.gson.JsonObject;
28  import com.google.gson.Strictness;
29  
30  import org.apache.commons.codec.binary.Base64;
31  import org.joda.time.DateTime;
32  import org.joda.time.DateTimeZone;
33  
34  import java.nio.charset.StandardCharsets;

36  /**
37   * A Utility class to encode / decode Policy files from and to Base 64 and Json.
38   */
39  public final class PolicyUtils {
40    /**
41     * Parses policies strictly, like {@code GsonUtil} in opencast-common does. That class is not used here to keep this
42     * module's dependencies small.
43     */
44    private static final Gson GSON = new GsonBuilder().setStrictness(Strictness.STRICT).create();
45    /** The JSON key for object that contains the date and ip conditions for the resource. */
46    private static final String CONDITION_KEY = "Condition";
47    /** The JSON key for the date and time the resource will become available. */
48    private static final String DATE_GREATER_THAN_KEY = "DateGreaterThan";
49    /** The JSON key for the date and time the resource will no longer be available. */
50    private static final String DATE_LESS_THAN_KEY = "DateLessThan";
51    /** The JSON key for the IP address of the acceptable client. */
52    private static final String IP_ADDRESS_KEY = "IpAddress";
53    /** The JSON key for the base url for the resource. */
54    private static final String RESOURCE_KEY = "Resource";
55    /** The JSON key for the main object of the policy. */
56    private static final String STATEMENT_KEY = "Statement";
57  
58    private PolicyUtils() {
59  
60    }
61  
62    /**
63     * Encode a {@link String} into Base 64 encoding
64     *
65     * @param value
66     *          The {@link String} to encode into base 64 encoding
67     * @return The {@link String} encoded into base 64.
68     */
69    public static String base64Encode(String value) {
70      return Base64.encodeBase64URLSafeString(value.getBytes());
71    }
72  
73    /**
74     * Decode a {@link String} from Base 64 encoding
75     *
76     * @param value
77     *          The {@link String} to encode into Base 64
78     * @return The {@link String} decoded from base 64.
79     */
80    public static String base64Decode(String value) {
81      return new String(Base64.decodeBase64(value), StandardCharsets.UTF_8);
82    }
83  
84    /**
85     * Get a {@link Policy} from JSON data.
86     *
87     * @param policyJson
88     *          The {@link String} representation of the json.
89     * @return A new {@link Policy} object populated from the JSON.
90     */
91    public static Policy fromJson(String policyJson) {
92      JsonObject jsonPolicy = GSON.fromJson(policyJson, JsonObject.class);
93  
94      JsonObject statement = jsonPolicy.getAsJsonObject(STATEMENT_KEY);
95      String resource = statement.get(RESOURCE_KEY).getAsString();
96      JsonObject condition = statement.getAsJsonObject(CONDITION_KEY);
97  
98      final String lessThanString = condition.get(DATE_LESS_THAN_KEY).getAsString();
99      final DateTime dateLessThan = new DateTime(Long.parseLong(lessThanString), DateTimeZone.UTC);
100 
101     final DateTime dateGreaterThan;
102     if (condition.has(DATE_GREATER_THAN_KEY)) {
103       dateGreaterThan = new DateTime(Long.parseLong(condition.get(DATE_GREATER_THAN_KEY).getAsString()),
104               DateTimeZone.UTC);
105     } else {
106       dateGreaterThan = null;
107     }
108 
109     final String ipAddress = condition.has(IP_ADDRESS_KEY) ? condition.get(IP_ADDRESS_KEY).getAsString() : null;
110 
111     return Policy.mkPolicyValidFromWithIP(resource, dateLessThan, dateGreaterThan, ipAddress);
112   }
113 
114   /**
115    * Render a {@link Policy} into JSON.
116    * <p>
117    * The exact bytes of this output are part of Opencast's contract with anything that has ever signed a URL:
118    * {@link #getPolicySignature(Policy, String)} takes a digest of this string, and
119    * <code>ResourceRequestUtil</code> verifies a request by re-rendering the decoded policy and comparing digests.
120    * Re-ordering a key or changing how a character is escaped therefore invalidates every signed URL already in
121    * circulation, and breaks third-party signers that reproduce this format.
122    * <p>
123    * That is why this is written out by hand rather than handed to a JSON library: the format is a fixed contract, so
124    * it should not be able to drift when a library is upgraded or replaced. <code>PolicyUtilsTest</code> pins the
125    * result.
126    *
127    * @param policy
128    *          The {@link Policy} to render into JSON.
129    * @return The JSON representation of the {@link Policy}.
130    */
131   public static String toJson(Policy policy) {
132     StringBuilder json = new StringBuilder(128);
133 
134     json.append("{\"").append(STATEMENT_KEY).append("\":{\"").append(CONDITION_KEY).append("\":{");
135     if (policy.getValidFrom().isPresent()) {
136       json.append('"').append(DATE_GREATER_THAN_KEY).append("\":")
137           .append(policy.getValidFrom().get().getMillis()).append(',');
138     }
139     json.append('"').append(DATE_LESS_THAN_KEY).append("\":").append(policy.getValidUntil().getMillis());
140     if (policy.getClientIpAddress().isPresent()) {
141       json.append(",\"").append(IP_ADDRESS_KEY).append("\":\"")
142           .append(escape(policy.getClientIpAddress().get().getHostAddress())).append('"');
143     }
144     json.append("},\"").append(RESOURCE_KEY).append("\":\"").append(escape(policy.getResource())).append("\"}}");
145 
146     return json.toString();
147   }
148 
149   /**
150    * Escape a string for inclusion in the policy JSON.
151    * <p>
152    * This deliberately reproduces the escaping Opencast has emitted since the signing format was introduced, which
153    * includes escaping the forward slash and the U+2000-U+20FF block. Both are optional per the JSON specification,
154    * but they are part of the signed bytes, so they cannot be dropped. See {@link #toJson(Policy)}.
155    *
156    * @param value
157    *          The string to escape.
158    * @return The escaped string, without surrounding quotes.
159    */
160   private static String escape(String value) {
161     StringBuilder escaped = new StringBuilder(value.length() + 16);
162     for (int i = 0; i < value.length(); i++) {
163       char ch = value.charAt(i);
164       switch (ch) {
165         case '"':
166           escaped.append("\\\"");
167           break;
168         case '\\':
169           escaped.append("\\\\");
170           break;
171         case '/':
172           escaped.append("\\/");
173           break;
174         case '\b':
175           escaped.append("\\b");
176           break;
177         case '\f':
178           escaped.append("\\f");
179           break;
180         case '\n':
181           escaped.append("\\n");
182           break;
183         case '\r':
184           escaped.append("\\r");
185           break;
186         case '\t':
187           escaped.append("\\t");
188           break;
189         default:
190           if (ch <= 0x1F || (ch >= 0x7F && ch <= 0x9F) || (ch >= 0x2000 && ch <= 0x20FF)) {
191             escaped.append(String.format("\\u%04X", (int) ch));
192           } else {
193             escaped.append(ch);
194           }
195           break;
196       }
197     }
198     return escaped.toString();
199   }

201   /**
202    * Create a {@link Policy} in Json format and Base 64 encoded.
203    *
204    * @param encodedPolicy
205    *          The String representation of the {@link Policy} in Json format and encoded into Base 64
206    * @return The {@link Policy} data
207    */
208   public static Policy fromBase64EncodedPolicy(String encodedPolicy) {
209     String decodedPolicyString = base64Decode(encodedPolicy);
210     return fromJson(decodedPolicyString);
211   }
212 
213   /**
214    * Create a {@link Policy} in Json format and Base 64 encoded.
215    *
216    * @param policy
217    *          The String representation of the {@link Policy} in Json format and encoded into Base 64
218    * @return The {@link Policy} data
219    */
220   public static String toBase64EncodedPolicy(Policy policy) {
221     return base64Encode(PolicyUtils.toJson(policy));
222   }
223 
224   /**
225    * Get an encrypted version of a {@link Policy} to use as a signature.
226    *
227    * @param policy
228    *          {@link Policy} that needs to be encrypted.
229    * @param encryptionKey
230    *          The key to use to encrypt the {@link Policy}.
231    * @return An encrypted version of the {@link Policy} that is also Base64 encoded to make it safe to transmit as a
232    *         query parameter.
233    * @throws Exception
234    *           Thrown if there is a problem encrypting or encoding the {@link Policy}
235    */
236   public static String getPolicySignature(Policy policy, String encryptionKey) throws Exception {
237     return SHA256Util.digest(PolicyUtils.toJson(policy), encryptionKey);
238   }
