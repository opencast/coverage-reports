43  /**
44   * Tests that the workflow operation worker leaves the values of operation configuration keys declared as templates by
45   * the handler untouched, and substitutes all other values as before.
46   */
47  public class WorkflowOperationWorkerTemplatedConfigurationTest {
48  
49    /** A workflow operation handler that declares which of its configuration keys are templates */
50    private static class TemplatedConfigurationHandler extends AbstractWorkflowOperationHandler {
51      private final Set<String> templatedKeys;
52  
53      TemplatedConfigurationHandler(Set<String> templatedKeys) {
54        super.id = "templated-configuration-test";
55        this.templatedKeys = templatedKeys;
56      }
57  
58      @Override
59      public Set<String> getTemplatedConfigurationKeys() {
60        return templatedKeys;
61      }
62  
63      @Override
64      public WorkflowOperationResult start(WorkflowInstance workflowInstance, JobContext context) {
65        return createResult(Action.CONTINUE);
66      }
67    }
68  
69    /**
70     * Builds a workflow instance with the given operation configurations and a single instantiable operation.
71     */
72    private WorkflowInstance buildWorkflow(Map<String, String> operationConfigurations) {
73      WorkflowInstance workflow = new WorkflowInstance();
74      workflow.setId(1);
75      workflow.setConfiguration("event_id", "my-event");
76      WorkflowOperationInstance operation = new WorkflowOperationInstance("operation", OperationState.INSTANTIATED);
77      operationConfigurations.forEach(operation::setConfiguration);
78      workflow.setOperations(List.of(operation));
79      return workflow;
80    }
81  
82    /**
83     * Builds a worker that runs the given handler without persisting the workflow.
84     */
85    private WorkflowOperationWorker buildWorker(WorkflowOperationHandler handler, WorkflowInstance workflow) {
86      WorkflowServiceImpl service = new WorkflowServiceImpl() {
87        @Override
88        public void update(WorkflowInstance workflowInstance) {
89          // no-op: do not persist
90        }
91      };
92      service.setSecurityService(new SecurityServiceStub());
93      return new WorkflowOperationWorker(handler, workflow, null, service);
94    }
95  
96    @Test
97    public void templatedConfigurationValuesAreNotSubstituted() throws Exception {
98      Map<String, String> operationConfigurations = new HashMap<>();
99      operationConfigurations.put("templated", "${event_id}");
100     operationConfigurations.put("plain", "${event_id}");
101     operationConfigurations.put("org", "${org_logo_small}");
102     WorkflowInstance workflow = buildWorkflow(operationConfigurations);
103     WorkflowOperationInstance operation = workflow.getCurrentOperation();
104 
105     WorkflowOperationWorker worker =
106         buildWorker(new TemplatedConfigurationHandler(Set.of("templated")), workflow);
107     worker.start();
108 
109     // The value of the templated key is left untouched for the handler to render
110     assertEquals("${event_id}", operation.getConfiguration("templated"));
111     // All other values are substituted as before
112     assertEquals("my-event", operation.getConfiguration("plain"));
113     assertEquals(
114         DefaultOrganization.DEFAULT_PROPERTIES.get("logo_small"),
115         operation.getConfiguration("org"));
116   }
117 
118   @Test
119   public void aNullTemplatedConfigurationKeySetIsTreatedAsEmpty() throws Exception {
120     Map<String, String> operationConfigurations = new HashMap<>();
121     operationConfigurations.put("templated", "${event_id}");
122     WorkflowInstance workflow = buildWorkflow(operationConfigurations);
123     WorkflowOperationInstance operation = workflow.getCurrentOperation();
124 
125     WorkflowOperationWorker worker = buildWorker(new TemplatedConfigurationHandler(null), workflow);
126     worker.start();
127 
128     assertEquals("my-event", operation.getConfiguration("templated"));
129   }
130 
131   @Test
132   public void allConfigurationValuesAreSubstitutedByDefault() throws Exception {
133     Map<String, String> operationConfigurations = new HashMap<>();
134     operationConfigurations.put("templated", "${event_id}");
135     operationConfigurations.put("plain", "${event_id}");
136     WorkflowInstance workflow = buildWorkflow(operationConfigurations);
137     WorkflowOperationInstance operation = workflow.getCurrentOperation();
138 
139     WorkflowOperationWorker worker =
140         buildWorker(new TemplatedConfigurationHandler(Set.of()), workflow);
141     worker.start();
142 
143     assertEquals("my-event", operation.getConfiguration("templated"));
144     assertEquals("my-event", operation.getConfiguration("plain"));
145   }
146 }
