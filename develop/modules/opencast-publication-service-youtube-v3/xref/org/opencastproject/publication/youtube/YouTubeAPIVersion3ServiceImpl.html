63    public void initialize(final ClientCredentials credentials) throws IOException {
64      final OAuth2CredentialFactory factory = new OAuth2CredentialFactoryImpl();

66      youTube = new YouTube.Builder(new NetHttpTransport(), new JacksonFactory(), credential).build();

69    @Override
70    public Video addVideoToMyChannel(final VideoUpload videoUpload) throws IOException {
71      final Video videoObjectDefiningMetadata = new Video();
72      final VideoStatus status = new VideoStatus();
73      status.setPrivacyStatus(videoUpload.getPrivacyStatus());
74      videoObjectDefiningMetadata.setStatus(status);
75      // Metadata lives in VideoSnippet
76      final VideoSnippet snippet = new VideoSnippet();
77      snippet.setTitle(videoUpload.getTitle());
78      snippet.setDescription(videoUpload.getDescription());
79      final String[] tags = videoUpload.getTags();
80      if (ArrayUtils.isNotEmpty(tags)) {
81        snippet.setTags(Arrays.asList(tags));
82      }
83      // Attach metadata to video object.
84      videoObjectDefiningMetadata.setSnippet(snippet);
85  
86      final File videoFile = videoUpload.getVideoFile();
87      final BufferedInputStream inputStream = new BufferedInputStream(new FileInputStream(videoFile));
88      final InputStreamContent mediaContent = new InputStreamContent("video/*", inputStream);
89      mediaContent.setLength(videoFile.length());
90  
91      final YouTube.Videos.Insert videoInsert = youTube.videos()
92          .insert("snippet,statistics,status", videoObjectDefiningMetadata, mediaContent);
93      final MediaHttpUploader uploader = videoInsert.getMediaHttpUploader();
94  
95      uploader.setDirectUploadEnabled(false);
96      uploader.setProgressListener(videoUpload.getProgressListener());
97      return execute(videoInsert);
98    }

100   @Override
101   public Playlist createPlaylist(final String title, final String description, PrivacyStatus privacyStatus,
102           final String... tags) throws IOException {
103     final PlaylistSnippet playlistSnippet = new PlaylistSnippet();
104     playlistSnippet.setTitle(title);
105     playlistSnippet.setDescription(description);
106     if (tags.length > 0) {
107       playlistSnippet.setTags(Arrays.asList(tags));
108     }
109     final PlaylistStatus playlistStatus = new PlaylistStatus();
110     playlistStatus.setPrivacyStatus(privacyStatus.name().toLowerCase());
111 
112     // Create playlist with metadata and status.
113     final Playlist youTubePlaylist = new Playlist();
114     youTubePlaylist.setSnippet(playlistSnippet);
115     youTubePlaylist.setStatus(playlistStatus);
116 
117     // The first argument tells the API what to return when a successful insert has been executed.
118     final YouTube.Playlists.Insert command = youTube.playlists().insert("snippet,status", youTubePlaylist);
119     return execute(command);
120   }

122   @Override
123   public PlaylistItem addPlaylistItem(final String playlistId, final String videoId) throws IOException {
124     // Resource type (video,playlist,channel) needs to be set along with resource id.
125     final ResourceId resourceId = new ResourceId();
126     resourceId.setKind("youtube#video");
127     resourceId.setVideoId(videoId);
128 
129     // Set the required snippet properties.
130     final PlaylistItemSnippet playlistItemSnippet = new PlaylistItemSnippet();
131     playlistItemSnippet.setTitle("First video in the test playlist");
132     playlistItemSnippet.setPlaylistId(playlistId);
133     playlistItemSnippet.setResourceId(resourceId);
134 
135     // Create the playlist item.
136     final PlaylistItem playlistItem = new PlaylistItem();
137     playlistItem.setSnippet(playlistItemSnippet);
138 
139     // The first argument tells the API what to return when a successful insert has been executed.
140     final YouTube.PlaylistItems.Insert playlistItemsInsertCommand
141         = youTube.playlistItems().insert("snippet,contentDetails", playlistItem);
142     return execute(playlistItemsInsertCommand);
143   }

164   @Override
165   public Video getVideoById(final String videoId) throws IOException {
166     final YouTube.Videos.List search = youTube.videos().list("id,snippet");
167     search.setId(videoId);
168     search.setFields("items(id,kind,snippet),nextPageToken,pageInfo,prevPageToken,tokenPagination");
169     final VideoListResponse response = execute(search);
170     return response.getItems().isEmpty() ? null : response.getItems().get(0);
171   }

173   @Override
174   public Playlist getMyPlaylistByTitle(final String title) throws IOException {
175     final String trimmedTitle = title.trim();
176     boolean searchedAllPlaylist = false;
177     Playlist playlist = null;
178     String nextPageToken = null;
179     while (!searchedAllPlaylist) {
180       final PlaylistListResponse searchResult = getMyPlaylists(nextPageToken, 50);
181       for (final Playlist p : searchResult.getItems()) {
182         if (p.getSnippet().getTitle().trim().equals(trimmedTitle)) {
183           playlist = p;
184           break;
185         }
186       }
187       nextPageToken = searchResult.getNextPageToken();
188       searchedAllPlaylist = nextPageToken == null;
189     }
190     return playlist;
191   }

193   @Override
194   public PlaylistListResponse getMyPlaylists(final String pageToken, final long maxResults) throws IOException {
195     final YouTube.Playlists.List search = youTube.playlists().list("id,snippet");
196     if (pageToken != null) {
197       search.set("pageToken", pageToken);
198     }
199     search.setMaxResults(maxResults);
200     search.setMine(true);
201     search.setFields("items(id,snippet),kind,nextPageToken,pageInfo,prevPageToken,tokenPagination");
202     return execute(search);
203   }

219   /**
220    * @param playlistId may not be {@code null}
221    * @param videoId may not be {@code null}
222    * @return null when not found.
223    * @throws IOException when transaction fails.
224    */
225   private PlaylistItem findPlaylistItem(final String playlistId, final String videoId) throws IOException {
226     final List<PlaylistItem> playlistItems = getAllPlaylistItems(playlistId);
227     PlaylistItem playlistItem = null;
228     for (final PlaylistItem next : playlistItems) {
229       final String id = next.getSnippet().getResourceId().getVideoId();
230       if (videoId.equals(id)) {
231         playlistItem = next;
232         break;
233       }
234     }
235     return playlistItem;
236   }
237 
238   /**
239    *
240    * @param playlistId may not be {@code null}
241    * @return zero or more playlist items.
242    * @throws IOException  when transaction fails.
243    */
244   private List<PlaylistItem> getAllPlaylistItems(final String playlistId) throws IOException {
245     final List<PlaylistItem> playlistItems = new LinkedList<PlaylistItem>();
246     boolean done = false;
247     String nextPageToken = null;
248     while (!done) {
249       final PlaylistItemListResponse searchResult = getPlaylistItems(playlistId, nextPageToken, 50);
250       playlistItems.addAll(searchResult.getItems());
251       nextPageToken = searchResult.getNextPageToken();
252       done = nextPageToken == null;
253     }
254     return playlistItems;
255   }

262    * @throws IOException  when transaction fails.
263    */
264   private <T> T execute(final YouTubeRequest<T> command) throws IOException {
265     return command.execute();
