22  package org.opencastproject.lifecyclemanagement.impl;
23  
24  import org.opencastproject.lifecyclemanagement.api.Action;
25  import org.opencastproject.lifecyclemanagement.api.LifeCyclePolicy;
26  import org.opencastproject.lifecyclemanagement.api.LifeCycleService;
27  import org.opencastproject.lifecyclemanagement.api.LifeCycleServiceException;
28  import org.opencastproject.lifecyclemanagement.api.LifeCycleTask;
29  import org.opencastproject.lifecyclemanagement.api.Status;
30  import org.opencastproject.util.NotFoundException;
31  
32  import org.slf4j.Logger;
33  import org.slf4j.LoggerFactory;
34  
35  /** Shared helper for {@link PolicyCheckRunner} and {@link RepeatingPolicyRunner}. */
36  final class LifeCycleTaskUtils {
37  
38    private static final Logger logger = LoggerFactory.getLogger(LifeCycleTaskUtils.class);
39  
40    private LifeCycleTaskUtils() {
41    }
42  
43    /**
44     * Creates a scheduled {@link LifeCycleTask} for the given policy and target entity, unless a task for that
45     * policy and entity already exists.
46     *
47     * @param lifeCycleService the service used to look up and persist tasks
48     * @param policy the policy the task is created for
49     * @param entityId the id of the entity the task will act on
50     * @throws LifeCycleServiceException If something went wrong in the lifecycle service
51     */
52    static void createTaskIfNotExists(LifeCycleService lifeCycleService, LifeCyclePolicy policy, String entityId)
53            throws LifeCycleServiceException {
54      try {
55        LifeCycleTask existingTask = lifeCycleService.getLifeCycleTaskByTargetId(entityId);
56        if (existingTask.getLifeCyclePolicyId().equals(policy.getId())) {
57          // Task already exists, skip creating one
58          return;
59        }
60      } catch (NotFoundException e) {
61        // Task does not exist yet, so create one
62      }
63  
64      LifeCycleTask task;
65      if (policy.getAction() == Action.START_WORKFLOW) {
66        task = new LifeCycleTaskStartWorkflow();
67      } else {
68        task = new LifeCycleTaskImpl();
69      }
70  
71      task.setLifeCyclePolicyId(policy.getId());
72      task.setTargetId(entityId);
73      task.setStatus(Status.SCHEDULED);
74  
75      lifeCycleService.createLifeCycleTask(task);
76      logger.debug("Created task based on policy " + policy.getTitle());
77    }
78  }
