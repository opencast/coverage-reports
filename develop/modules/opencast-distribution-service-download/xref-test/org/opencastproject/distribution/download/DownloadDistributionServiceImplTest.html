368   /**
369    * An element whose publication URI refers to the channel and media package of the current
370    * job resolves to the file in that channel's subtree
371    */
372   @Test
373   public void testGetDistributionFileOfCurrentJob() throws Exception {
374     String orgId = defaultOrganization.getId();
375     String mpId = mp.getIdentifier().toString();
376 
377     // Five-segment layout: .../{orgId}/{channelId}/{mpId}/{elementId}/{fileName}
378     MediaPackageElement element = trackElement("track-1",
379             UrlSupport.concat(UrlSupport.DEFAULT_BASE_URL, orgId, "engage-player", mpId, "track-1", "media.mov"));
380     File file = service.getDistributionFile("engage-player", mp, element);
381     File expected = new File(distributionRoot,
382             Paths.get(orgId, "engage-player", mpId, "track-1", "media.mov").toString());
383     Assert.assertEquals(expected, file);
384 
385     // The element segment of the URI is not compared against the element identifier: a
386     // distributed element is a copy of the original element whose publication URI references
387     // the original element's identifier, and adding the copy to a media package that still
388     // contains the original element generates a new identifier for the copy. Retracting such
389     // a renamed element must still find its file.
390     MediaPackageElement renamedElement = trackElement("renamed-id",
391             UrlSupport.concat(UrlSupport.DEFAULT_BASE_URL, orgId, "engage-player", mpId, "track-1", "media.mov"));
392     Assert.assertEquals(expected, service.getDistributionFile("engage-player", mp, renamedElement));
393 
394     // Legacy layout without channel level: .../{orgId}/{mpId}/{elementId}/{fileName}
395     MediaPackageElement legacyElement = trackElement("track-1",
396             UrlSupport.concat(UrlSupport.DEFAULT_BASE_URL, orgId, mpId, "track-1", "media.mov"));
397     File legacyFile = service.getDistributionFile("engage-player", mp, legacyElement);
398     File legacyExpected = new File(distributionRoot,
399             Paths.get(orgId, mpId, "track-1", "media.mov").toString());
400     Assert.assertEquals(legacyExpected, legacyFile);
401   }

403   /**
404    * An element whose publication URI refers to another channel or media package must not be
405    * able to address that channel's or media package's files, nor may the URI use unsafe path
406    * segments. A URI that stays within the channel and media package of the current job but
407    * names a different element is deliberately accepted (see
408    * {@link #testGetDistributionFileOfCurrentJob()})
409    */
410   @Test
411   public void testGetDistributionFileRejectsForeignUri() throws Exception {
412     String orgId = defaultOrganization.getId();
413     String mpId = mp.getIdentifier().toString();
414 
415     // Five-segment layout with a foreign channel or media package
416     assertIllegalState(
417             () -> service.getDistributionFile("engage-player", mp,
418                     trackElement("track-1", UrlSupport.concat(UrlSupport.DEFAULT_BASE_URL, orgId, "oai-pmh", mpId,
419                             "track-1", "media.mov"))),
420             "URI of another channel");
421     assertIllegalState(
422             () -> service.getDistributionFile("engage-player", mp,
423                     trackElement("track-1", UrlSupport.concat(UrlSupport.DEFAULT_BASE_URL, orgId,
424                             "engage-player", "other-mp", "track-1", "media.mov"))),
425             "URI of another media package");
426 
427     // Legacy layout with a foreign media package
428     assertIllegalState(
429             () -> service.getDistributionFile("engage-player", mp,
430                     trackElement("track-1",
431                             UrlSupport.concat(UrlSupport.DEFAULT_BASE_URL, orgId, "other-mp", "track-1",
432                                     "media.mov"))),
433             "legacy URI of another media package");
434 
435     // Unsafe path segments
436     assertIllegalState(
437             () -> service.getDistributionFile("engage-player", mp,
438                     trackElement("track-1", UrlSupport.concat(UrlSupport.DEFAULT_BASE_URL, orgId,
439                             "engage-player", mpId, "..", "media.mov"))),
440             "traversal segment in URI");
441     assertIllegalState(
442             () -> service.getDistributionFile("engage-player", mp,
443                     trackElement("track-1", UrlSupport.DEFAULT_BASE_URL + "/" + orgId + "//engage-player/" + mpId
444                             + "/track-1/media.mov")),
445             "empty segment in URI");
446   }

464   /**
465    * A publication URI in the legacy layout without channel level is only compared with the
466    * media package of the job. If that media package is named like a channel directory, such
467    * a URI can address the element directory of another media package in that channel. A
468    * retraction must not delete it (nor its parent, the other media package's directory).
469    */
470   @Test
471   public void testRetractLegacyUriDoesNotDeleteForeignElementDirectory() throws Exception {
472     String orgId = defaultOrganization.getId();
473     File victimFile = new File(distributionRoot,
474             Paths.get(orgId, "engage-player", "victim-mp", "victim-element", "media.mov").toString());
475     FileUtils.forceMkdir(victimFile.getParentFile());
476     FileUtils.writeStringToFile(victimFile, "must not be deleted", "UTF-8");
477 
478     MediaPackage channelNamedMp = MediaPackageParser.getFromXml(
479             "<mediapackage xmlns=\"http://mediapackage.opencastproject.org\" id=\"engage-player\"/>");
480     MediaPackageElement element = trackElement("track-1",
481             UrlSupport.concat(UrlSupport.DEFAULT_BASE_URL, orgId, "engage-player", "victim-mp", "victim-element"));
482     try {
483       service.retractElement("internal", channelNamedMp, element);
484       Assert.fail("Expected the retraction of a directory to be rejected");
485     } catch (DistributionException e) {
486       // intended
487       Assert.assertTrue("The element of another media package must not be deleted", victimFile.isFile());
488     } finally {
489       FileUtils.deleteQuietly(victimFile.getParentFile().getParentFile());
490     }
491   }

494     String xml = "<track xmlns=\"http://mediapackage.opencastproject.org\" id=\"" + id
495             + "\" type=\"presenter/source\"><url>" + url + "</url></track>";
496     return MediaPackageElementParser.getFromXml(xml);

499   private static void assertIllegalState(ThrowingRunnable runnable, String description) {
500     try {
501       runnable.run();
502       Assert.fail(description + " must be rejected");
503     } catch (IllegalStateException e) {
504       // intended
505     } catch (Exception e) {
506       Assert.fail(description + " was rejected with " + e + " instead of IllegalStateException");
507     }
508   }
