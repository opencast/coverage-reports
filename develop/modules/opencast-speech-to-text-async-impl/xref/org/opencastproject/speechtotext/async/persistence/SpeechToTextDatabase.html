64    public void activate(ComponentContext cc) {

84    public SpeechToTextControl storeSpeechToTextControl(String mediaPackageId, long workflowId, JpaJob job)
85            throws SpeechToTextAsyncException {
86      Date created = job.getDateCreated() != null ? job.getDateCreated() : new Date();
87      return storeSpeechToTextControl(
88              new SpeechToTextControl(mediaPackageId, workflowId, job, created, SpeechToTextControl.Status.InProgress));
89    }
90  
91    protected SpeechToTextControl storeSpeechToTextControl(SpeechToTextControl entity) throws SpeechToTextAsyncException {
92      try {
93        return db.execTx(em -> {
94          em.persist(entity);
95          return entity;
96        });
97      } catch (Exception e) {
98        throw new SpeechToTextAsyncException(e);
99      }
100   }
101 
102   public SpeechToTextControl findByJob(JpaJob job) throws SpeechToTextAsyncException {
103     try {
104       return db.exec(em -> {
105         return namedQuery.findOpt("SpeechToTextControl.findByJob", SpeechToTextControl.class, Pair.of("job", job))
106                 .apply(em).orElse(null);
107       });
108     } catch (Exception e) {
109       throw new SpeechToTextAsyncException(e);
110     }
111   }

113   /**
114    * Returns a list of STT controls that are in one of the status passed.
115    *
116    * @param status
117    *          Status list to search
118    */
119   public List<SpeechToTextControl> findByStatus(SpeechToTextControl.Status... status)
120           throws SpeechToTextAsyncException {
121     Collection<SpeechToTextControl.Status> statusCol = Arrays.stream(status)
122             .collect(Collectors.toCollection(HashSet::new));
123     try {
124       return db.exec(em -> {
125         return namedQuery
126                 .findAll("SpeechToTextControl.findByStatus", SpeechToTextControl.class, Pair.of("status", statusCol))
127                 .apply(em);
128       });
129     } catch (Exception e) {
130       throw new SpeechToTextAsyncException(e);
131     }
132   }

180   public int updateStatusByJob(SpeechToTextControl.Status status, JpaJob... jobs) throws SpeechToTextAsyncException {
181     try {
182       return db.execTx(em -> {
183         Query q = em.createNamedQuery("SpeechToTextControl.updateStatusByJob");
184         q.setParameter("status", status);
185         q.setParameter("jobs", Arrays.asList(jobs));
186         return q.executeUpdate();
187       });
188     } catch (Exception e) {
189       throw new SpeechToTextAsyncException(e);
190     }
191   }

206   public int transitionStatusByDate(SpeechToTextControl.Status newStatus, Date olderThan,
207           SpeechToTextControl.Status... oldStatus) throws SpeechToTextAsyncException {
208     try {
209       logger.debug("Changing status of stt controls older than {} from {} to {}", olderThan, Arrays.toString(oldStatus),
210               newStatus);
211       return db.execTx(em -> {
212         Query q = em.createNamedQuery("SpeechToTextControl.updateStatusByStatusAndDate");
213         q.setParameter("oldStatus", Arrays.asList(oldStatus));
214         q.setParameter("newStatus", newStatus);
215         q.setParameter("date", olderThan);
216         int counter = q.executeUpdate();
217         logger.debug("Changed status of {} rows from {} to {}", counter, Arrays.toString(oldStatus), newStatus);
218         return counter;
219       });
220     } catch (Exception e) {
221       throw new SpeechToTextAsyncException(e);
222     }
223   }
