114   private enum TrackSelectionStrategy {
115     PRESENTER_OR_NOTHING,
116     PRESENTATION_OR_NOTHING,
117     TRY_PRESENTER_FIRST,
118     TRY_PRESENTATION_FIRST,
119     EVERYTHING;
120 
121     private static TrackSelectionStrategy fromString(String value) {
122       for (TrackSelectionStrategy strategy : values()) {
123         if (strategy.name().equalsIgnoreCase(value)) {
124           return strategy;
125         }
126       }
127       throw new IllegalArgumentException(
128           "No TrackSelectionStrategy enum constant " + TrackSelectionStrategy.class.getCanonicalName() + "." + value);
129     }
130   }

155   /**
156    * {@inheritDoc}
157    *
158    * @see
159    * org.opencastproject.workflow.api.WorkflowOperationHandler#start(org.opencastproject.workflow.api.WorkflowInstance,
160    * org.opencastproject.job.api.JobContext)
161    */
162   @Override
163   public WorkflowOperationResult start(WorkflowInstance workflowInstance, JobContext context)
164           throws WorkflowOperationException {
165 
166     MediaPackage mediaPackage = workflowInstance.getMediaPackage();
167     logger.info("Start speech-to-text workflow operation for media package {}", mediaPackage);
168 
169     // Defaults to `false` if `null`
170     var async = BooleanUtils.toBoolean(workflowInstance.getCurrentOperation().getConfiguration(ASYNCHRONOUS));
171 
172     ConfiguredTagsAndFlavors tagsAndFlavors = getTagsAndFlavors(workflowInstance,
173             Configuration.many, Configuration.one,
174             Configuration.many, Configuration.one);
175     MediaPackageElementFlavor sourceFlavor = tagsAndFlavors.getSingleSrcFlavor();
176     List<String> srcTags = tagsAndFlavors.getSrcTags();
177 
178     TrackSelector trackSelector = new TrackSelector();
179     trackSelector.addFlavor(sourceFlavor);
180     for (String tag : srcTags) {
181       trackSelector.addTag(tag);
182     }
183     Collection<Track> tracks = trackSelector.select(mediaPackage, true);
184 
185     if (tracks.isEmpty()) {
186       throw new WorkflowOperationException(
187               String.format("No tracks with source flavor '%s' found for transcription", sourceFlavor));
188     }
189 
190     logger.info("Found {} track(s) with source flavor '{}'.", tracks.size(), sourceFlavor);
191 
192     // Get the information in which language the audio track should be
193     String languageCode = getMediaPackageLanguage(mediaPackage, workflowInstance);
194 
195     // How to save the subtitle file? (as attachment, as track...)
196     AppendSubtitleAs appendSubtitleAs = howToAppendTheSubtitles(workflowInstance);
197 
198     // Translate to english
199     Boolean translate = getTranslationMode(workflowInstance);
200 
201     // Create sublist that includes only the tracks that has audio
202     List<Track> tracksWithAudio = tracks.stream().filter(Track::hasAudio).collect(Collectors.toList());
203 
204     // Get the track selection strategy from the workflow configuration
205     // If nothing is set, all tracks (with audio) will be transcribed
206     TrackSelectionStrategy trackSelectionStrategy = getTrackSelectionStrategy(mediaPackage, workflowInstance);
207 
208     // Use the selection strategy from the workflow config to get the tracks we want to transcribe
209     List<Track> tracksToTranscribe = filterTracksByStrategy(tracksWithAudio, trackSelectionStrategy);
210     if (tracksToTranscribe.isEmpty()) {
211       logger.info("No subtitles were created for media package {}. "
212           + "Workflow Configuration 'track-selection-strategy' is set to {}", mediaPackage, trackSelectionStrategy);
213       return createResult(mediaPackage, WorkflowOperationResult.Action.SKIP);
214     }
215 
216     // Load the 'limit-to-one' configuration from the workflow operation.
217     // This configuration sets the limit of generated subtitle files to one
218     boolean limitToOne = BooleanUtils.toBoolean(workflowInstance.getCurrentOperation().getConfiguration(LIMIT_TO_ONE));
219     if (limitToOne) {
220       tracksToTranscribe = List.of(tracksToTranscribe.get(0));
221     }
222 
223     if (async) {
224       createSubtitleAsync(workflowInstance, tracksToTranscribe, languageCode, translate);
225     } else {
226       for (Track track : tracksToTranscribe) {
227         createSubtitle(track, languageCode, mediaPackage, tagsAndFlavors, appendSubtitleAs, translate);
228       }
229     }
230 
231     logger.info("Speech-To-Text workflow operation for media package {} completed", mediaPackage);
232     return createResult(mediaPackage, WorkflowOperationResult.Action.CONTINUE);
233   }

235   /**
236    * Filters the tracks by the strategy configured in the workflow operation
237    * @param tracksWithAudio        List of the tracks that includes audio
238    * @param trackSelectionStrategy The strategy configured in the workflow operation
239    * @return The filtered tracks
240    */
241   private List<Track> filterTracksByStrategy(List<Track> tracksWithAudio,
242       TrackSelectionStrategy trackSelectionStrategy) {
243 
244     List<Track> tracksToTranscribe = new ArrayList<>();
245     if (!tracksWithAudio.isEmpty()) {
246 
247       String presenterTypeConstant = MediaPackageElements.PRESENTER_SOURCE.getType();
248       String presentationTypeConstant = MediaPackageElements.PRESENTATION_SOURCE.getType();
249 
250       // Creates a sublist only including the presenter tracks
251       List<Track> presenterTracksWithAudio = tracksWithAudio.stream()
252           .filter(track -> Objects.equals(track.getFlavor().getType(), presenterTypeConstant))
253           .collect(Collectors.toList());
254 
255       // Creates a sublist only including the presentation tracks
256       List<Track> presentationTracksWithAudio = tracksWithAudio.stream()
257           .filter(track -> Objects.equals(track.getFlavor().getType(), presentationTypeConstant))
258           .collect(Collectors.toList());
259 
260       if (TrackSelectionStrategy.PRESENTER_OR_NOTHING.equals(trackSelectionStrategy)) {
261         tracksToTranscribe.addAll(presenterTracksWithAudio);
262       }
263 
264       if (TrackSelectionStrategy.PRESENTATION_OR_NOTHING.equals(trackSelectionStrategy)) {
265         tracksToTranscribe.addAll(presentationTracksWithAudio);
266       }
267 
268       if (TrackSelectionStrategy.TRY_PRESENTER_FIRST.equals(trackSelectionStrategy)) {
269         tracksToTranscribe.addAll(presenterTracksWithAudio);
270         if (tracksToTranscribe.isEmpty()) {
271           tracksToTranscribe.addAll(tracksWithAudio);
272         }
273       }
274 
275       if (TrackSelectionStrategy.TRY_PRESENTATION_FIRST.equals(trackSelectionStrategy)) {
276         tracksToTranscribe.addAll((presentationTracksWithAudio));
277         if (tracksToTranscribe.isEmpty()) {
278           tracksToTranscribe.addAll(tracksWithAudio);
279         }
280       }
281 
282       if (TrackSelectionStrategy.EVERYTHING.equals(trackSelectionStrategy)) {
283         tracksToTranscribe.addAll(tracksWithAudio);
284       }
285     }
286     return tracksToTranscribe;
287   }

289   /**
290    * Creates the subtitle file for a track and appends it to the media package.
291    *
292    * @param track The track from which the subtitles are created.
293    * @param languageCode The language of the track.
294    * @param parentMediaPackage The media package where the track is located.
295    * @param tagsAndFlavors Tags and flavors instance (to get target flavor information)
296    * @param appendSubtitleAs Tells how the subtitles file has to be appended.
297    * @param translate Enable translation to english.
298    * @throws WorkflowOperationException Get thrown if an error occurs.
299    */
300   private void createSubtitle(Track track, String languageCode, MediaPackage parentMediaPackage,
301           ConfiguredTagsAndFlavors tagsAndFlavors, AppendSubtitleAs appendSubtitleAs, Boolean translate)
302           throws WorkflowOperationException {
303 
304     // Start the transcription job, create subtitles file
305     URI trackURI = track.getURI();
306 
307     Job job;
308     logger.info("Generating subtitle for '{}'...", trackURI);
309     try {
310       job = speechToTextService.transcribe(trackURI, languageCode, translate, false);
311     } catch (SpeechToTextServiceException e) {
312       throw new WorkflowOperationException(
313               String.format("Generating subtitles for '%s' in media package '%s' failed",
314                       trackURI, parentMediaPackage), e);
315     }
316 
317     if (!waitForStatus(job).isSuccess()) {
318       throw new WorkflowOperationException(
319               String.format("Speech-to-Text job for media package '%s' failed", parentMediaPackage));
320     }
321 
322     // subtitles file is generated now, put it into the media package
323     try {
324       String[] jobOutput = job.getPayload().split(",");
325       URI output = new URI(jobOutput[0]);
326       String outputLanguage = jobOutput[1];
327       String engineType = jobOutput[2];
328 
329       MediaPackageElement subtitleMediaPackageElement;
330       switch (appendSubtitleAs) {
331         case attachment:
332           subtitleMediaPackageElement = new AttachmentImpl();
333           break;
334         case track:
335         default:
336           subtitleMediaPackageElement = new TrackImpl();
337       }
338 
339       subtitleMediaPackageElement.generateIdentifier();
340       try (InputStream in = workspace.read(output)) {
341         URI uri = workspace.put(parentMediaPackage.getIdentifier().toString(),
342                 subtitleMediaPackageElement.getIdentifier(),
343                 FilenameUtils.getName(output.getPath()), in);
344         subtitleMediaPackageElement.setURI(uri);
345       }
346       MediaPackageElementFlavor targetFlavor = tagsAndFlavors.getSingleTargetFlavor().applyTo(track.getFlavor());
347       subtitleMediaPackageElement.setFlavor(targetFlavor);
348 
349       ConfiguredTagsAndFlavors.TargetTags targetTags = tagsAndFlavors.getTargetTags();
350       targetTags.getOverrideTags().add("lang:" + outputLanguage);
351       targetTags.getOverrideTags().add("generator-type:auto");
352       targetTags.getOverrideTags().add("generator:" + engineType.toLowerCase());
353 
354       // this is used to set some values automatically, like the correct mimetype
355       Job inspection = mediaInspectionService.enrich(subtitleMediaPackageElement, true);
356       if (!waitForStatus(inspection).isSuccess()) {
357         throw new SpeechToTextServiceException(String.format(
358                 "Transcription for '%s' failed at enriching process", trackURI));
359       }
360 
361       subtitleMediaPackageElement = MediaPackageElementParser.getFromXml(inspection.getPayload());
362 
363       applyTargetTagsToElement(targetTags, subtitleMediaPackageElement);
364 
365       parentMediaPackage.add(subtitleMediaPackageElement);
366 
367       workspace.delete(output);
368     } catch (Exception e) {
369       throw new WorkflowOperationException("Error handling text-to-speech service output", e);
370     }
371 
372     try {
373       workspace.cleanup(parentMediaPackage.getIdentifier());
374     } catch (IOException e) {
375       throw new WorkflowOperationException(e);
376     }
377   }

379   /**
380    * Start the transcription, but don't actually wait for the process to finish. Instead, let the jobs run
381    * asynchronously and just store the launched jobs in the workflow configuration.
382    * @param workflow Workflow instance to store the jobs in
383    * @param tracks Tracks to run the transcription on
384    * @param languageCode Language to use
385    * @param translate If the transcription should be translated
386    * @throws WorkflowOperationException
387    */
388   private void createSubtitleAsync(WorkflowInstance workflow, List<Track> tracks, String languageCode,
389       Boolean translate) throws WorkflowOperationException {
390 
391     logger.info("Asynchronously generating subtitles");
392     StringBuilder jobs = new StringBuilder();
393     try {
394       try {
395         // Make sure workflow does not have any jobs being tracked currently e.g. cleanup a previous
396         // operation with retry-strategy=hold that failed.
397         speechToTextTracker.untrackAll(workflow.getId());
398       } catch (SpeechToTextAsyncException e) {
399         logger.warn("Could not delete existing trackers for mp {}", workflow.getMediaPackage().getIdentifier());
400       }
401       for (var track: tracks) {
402         // Make a copy of the track so that it isn't deleted by cleanup at the end of the workflow
403         String path = track.getURI().getPath();
404         String extension = path.lastIndexOf(".") > -1 ? path.substring(path.lastIndexOf(".")) : "";
405         String fileName = String.format("%d-%s%s", workflow.getId(), track.getIdentifier(), extension);
406         URI inputUri = workspace.putInCollection(STT_ASYNC_COLLECTION, fileName, workspace.read(track.getURI()));
407 
408         var job = speechToTextService.transcribe(inputUri, languageCode, translate, true);
409         jobs.append(",").append(job.getId());
410         try {
411           speechToTextTracker.track(job, workflow.getMediaPackage().getIdentifier().toString(), workflow.getId());
412         } catch (SpeechToTextAsyncException e) {
413           logger.warn("Could not add tracker to async speech to text process for mp {}, track {}",
414                   workflow.getMediaPackage().getIdentifier(), track, e);
415           // Do not fail the workflow, just not track it
416         }
417       }
418     } catch (IOException | NotFoundException | SpeechToTextServiceException e) {
419       throw new WorkflowOperationException(
420           String.format("Starting subtitle job in media package '%s' failed",
421               workflow.getMediaPackage().getIdentifier()), e);
422     }
423 
424     var config = Objects.toString(workflow.getConfiguration(JOBS_WORKFLOW_CONFIGURATION), "") + jobs;
425     workflow.setConfiguration(JOBS_WORKFLOW_CONFIGURATION, config.replaceFirst("^,", ""));
426   }

428   /**
429    * Get the config for the "track selection strategy". It's used to determine which tracks shall be transcribed.
430    * If there are 2 Videos and both has audio for example, what audio shall be transcribed?
431    *
432    * @param workflowInstance Contains the workflow configuration.
433    * @return Which strategy to use
434    * @throws WorkflowOperationException Get thrown if an error occurs.
435    */
436   private TrackSelectionStrategy getTrackSelectionStrategy(MediaPackage mediaPackage, WorkflowInstance workflowInstance)
437           throws WorkflowOperationException {
438 
439     WorkflowOperationInstance operation = workflowInstance.getCurrentOperation();
440     String strategyCfg = StringUtils.trimToEmpty(operation.getConfiguration(TRACK_SELECTION_STRATEGY)).toLowerCase();
441 
442     if (strategyCfg.isEmpty()) {
443       return TrackSelectionStrategy.EVERYTHING; // "transcribe everything" is the default/fallback
444     }
445     try {
446       return TrackSelectionStrategy.fromString(strategyCfg);
447     } catch (IllegalArgumentException e) {
448       throw new WorkflowOperationException(String.format(
449           "Speech-to-Text job for media package '%s' failed, because of wrong workflow configuration. "
450               + "track-selection-strategy of type '%s' does not exist.", mediaPackage, strategyCfg));
451     }
452   }

455   /**
456    * Get the information how to append the subtitles file to the media package.
457    *
458    * @param workflowInstance Contains the workflow configuration.
459    * @return How to append the subtitles file to the media package.
460    * @throws WorkflowOperationException Get thrown if an error occurs.
461    */
462   private AppendSubtitleAs howToAppendTheSubtitles(WorkflowInstance workflowInstance)
463           throws WorkflowOperationException {
464     WorkflowOperationInstance operation = workflowInstance.getCurrentOperation();
465     String targetElement = StringUtils.trimToEmpty(operation.getConfiguration(TARGET_ELEMENT)).toLowerCase();
466     if (targetElement.isEmpty()) {
467       return AppendSubtitleAs.track;
468     }
469     try {
470       return AppendSubtitleAs.valueOf(targetElement);
471     } catch (IllegalArgumentException e) {
472       throw new WorkflowOperationException(String.format(
473           "Speech-to-Text job for media package '%s' failed, because of wrong workflow configuration. "
474               + "target-element of type '%s' does not exist.", workflowInstance.getMediaPackage(), targetElement));
475     }
476   }

489   /**
490    * Searches some places to get the right language of the media package / track.
491    *
492    * @param mediaPackage The media package from which the subtitles are generated.
493    * @param workflowInstance Contains the workflow configuration.
494    * @return The language of the media package / track.
495    */
496   private String getMediaPackageLanguage(MediaPackage mediaPackage, WorkflowInstance workflowInstance) {
497 
498     // First look if there is a fixed language configured in the operation
499     WorkflowOperationInstance operation = workflowInstance.getCurrentOperation();
500     String language = StringUtils.trimToEmpty(operation.getConfiguration(LANGUAGE_CODE));
501 
502     if (language.isEmpty()) {
503       // If not we look in the dublin core metadata if the language is available
504       MediaPackageMetadata dublinCoreMetadata = dublinCoreCatalogService.getMetadata(mediaPackage);
505       language = StringUtils.trimToEmpty(dublinCoreMetadata.getLanguage());
506     }
507 
508     if (language.isEmpty()) {
509       // If there is still no language, we look in the media package itself
510       language = StringUtils.trimToEmpty(mediaPackage.getLanguage());
511     }
512 
513     if (language.isEmpty()) {
514       // Use the fallback language if the operation configuration defines one
515       language = Objects.toString(operation.getConfiguration(LANGUAGE_FALLBACK), "");
516     }
517 
518     return language;
519   }
