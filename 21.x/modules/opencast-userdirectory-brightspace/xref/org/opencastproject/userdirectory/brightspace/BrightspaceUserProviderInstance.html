124   @Activate
125   @Modified
126   public void updated(Map<String, Object> properties) throws ConfigurationException {
127     pid = (String) properties.get(Constants.SERVICE_PID);
128 
129     String organizationId = (String) properties.get(ORGANIZATION_KEY);
130     String urlStr = (String) properties.get(BRIGHTSPACE_URL);
131     String systemUserId = (String) properties.get(BRIGHTSPACE_USER_ID);
132     String systemUserKey = (String) properties.get(BRIGHTSPACE_USER_KEY);
133     final String applicationId = (String) properties.get(BRIGHTSPACE_APP_ID);
134     final String applicationKey = (String) properties.get(BRIGHTSPACE_APP_KEY);
135 
136     int cacheSize;
137     String cacheSizeStr = (String) properties.get(CACHE_SIZE_KEY);
138     if (StringUtils.isBlank(cacheSizeStr)) {
139       cacheSize = DEFAULT_CACHE_SIZE_VALUE;
140     } else {
141       cacheSize = NumberUtils.toInt(cacheSizeStr);
142     }
143 
144     int cacheExpiration;
145     String cacheExpirationStr = (String) properties.get(CACHE_EXPIRATION_KEY);
146     if (StringUtils.isBlank(cacheExpirationStr)) {
147       cacheExpiration = DEFAULT_CACHE_EXPIRATION_VALUE;
148     } else {
149       cacheExpiration = NumberUtils.toInt(cacheExpirationStr);
150     }
151 
152     String rolesStr = (String) properties.get(BRIGHTSPACE_INSTRUCTOR_ROLES_KEY);
153     if (StringUtils.isBlank(rolesStr)) {
154       rolesStr = DEFAULT_BRIGHTSPACE_INSTRUCTOR_ROLES;
155     }
156     instructorRoles = parsePropertyLineAsSet(rolesStr);
157     logger.debug("Brightspace instructor roles: {}", instructorRoles);
158 
159     String ignoredUsersStr = (String) properties.get(IGNORED_USERNAMES_KEY);
160     if (StringUtils.isBlank(ignoredUsersStr)) {
161       ignoredUsersStr = DEFAULT_IGNORED_USERNAMES;
162     }
163     ignoredUsernames = parsePropertyLineAsSet(ignoredUsersStr);
164     logger.debug("Ignored users: {}", ignoredUsernames);
165 
166     validateUrl(urlStr);
167     validateConfigurationKey(ORGANIZATION_KEY, organizationId);
168     validateConfigurationKey(BRIGHTSPACE_USER_ID, systemUserId);
169     validateConfigurationKey(BRIGHTSPACE_USER_KEY, systemUserKey);
170     validateConfigurationKey(BRIGHTSPACE_APP_ID, applicationId);
171     validateConfigurationKey(BRIGHTSPACE_APP_KEY, applicationKey);
172 
173     try {
174       organization = orgDirectory.getOrganization(organizationId);
175     } catch (NotFoundException nfe) {
176       logger.warn("Organization {} not found!", organizationId);
177       throw new ConfigurationException(ORGANIZATION_KEY, "not found");
178     }
179 
180     client = new BrightspaceClientImpl(urlStr, applicationId, applicationKey, systemUserId, systemUserKey);
181 
182     logger.info("Configured BrightspaceUserProviderInstance(pid={}, url={}, cacheSize={}, cacheExpiration={}, "
183                   + "InstructorRoles={}, ignoredUserNames={})", pid, client.getURL(), cacheSize, cacheExpiration,
184                   instructorRoles, ignoredUsernames);
185 
186     cache = CacheBuilder.newBuilder().maximumSize(cacheSize).expireAfterWrite(cacheExpiration, TimeUnit.MINUTES)
187             .build(new CacheLoader<String, Object>() {
188               @Override
189               public Object load(String username) {
190                 User user = loadUserFromBrightspace(username);
191                 return user == null ? nullToken : user;
192               }
193             });
194   }
195 
196   private void validateConfigurationKey(String key, String value) throws ConfigurationException {
197     if (StringUtils.isBlank(value)) {
198       throw new ConfigurationException(key, "is not set");
199     }
200   }
201 
202   private void validateUrl(String urlStr) throws ConfigurationException {
203     if (StringUtils.isBlank(urlStr)) {
204       throw new ConfigurationException(BRIGHTSPACE_URL, "is not set");
205     } else {
206       try {
207         new URI(urlStr);
208       } catch (URISyntaxException e) {
209         throw new ConfigurationException(BRIGHTSPACE_URL, "not a URL");
210       }
211     }
212   }

214   private Set<String> parsePropertyLineAsSet(String configLine) {
215     Set<String> set = new HashSet<>();
216     String[] configs = configLine.split(",");
217     for (String config: configs) {
218       set.add(config.trim());
219     }
220     return set;
221   }

248   @Override
249   public User loadUser(String userName) {
250     logger.debug("getting user from cache");
251 
252     try {
253       Object user = this.cache.getUnchecked(userName);
254       if (user != this.nullToken) {
255         logger.debug("Returning user {} from cache", userName);
256         return (User) user;
257       } else {
258         logger.debug("Returning null user from cache");
259         return null;
260       }
261     } catch (ExecutionError | UncheckedExecutionException ee) {
262       logger.warn("Exception while loading user {}", userName, ee);
263       return null;
264     }
265   }

302   @Override
303   public List<Role> getRolesForUser(String username) {
304     User user = this.loadUser(username);
305     if (user != null) {
306       logger.debug("Returning cached role set for {}", username);
307       return new ArrayList<>(user.getRoles());
308     }
309     logger.debug("Return empty role set for {} - not found in Brightspace", username);
310     return Collections.emptyList();
311   }

318   private User loadUserFromBrightspace(String username) {
319     if (this.cache == null) {
320       throw new IllegalStateException("The Brightspace user detail service has not yet been configured");
321     } else if (ignoredUsernames.stream().anyMatch(u -> u.equals(username))) {
322       logger.debug("We don't answer for: " + username);
323       return null;
324     } else {
325 
326       logger.debug("In loadUserFromBrightspace, currently processing user: {}", username);
327       JaxbOrganization jaxbOrganization = JaxbOrganization.fromOrganization(organization);
328 
329       Thread currentThread = Thread.currentThread();
330       ClassLoader originalClassloader = currentThread.getContextClassLoader();
331       BrightspaceUser brightspaceUser;
332 
333       try {
334         brightspaceUser = this.client.findUser(username);
335 
336         if (brightspaceUser != null) {
337           logger.info("Retrieved user {}", brightspaceUser.getUserId());
338           String brightspaceUserId = brightspaceUser.getUserId();
339 
340           List<String> roleList = client.getRolesFromBrightspace(brightspaceUserId, instructorRoles);
341           logger.debug("Brightspace user {} with id {} with roles: {}", username, brightspaceUserId, roleList);
342 
343           Set<JaxbRole> roles = new HashSet<>();
344           boolean isInstructor = false;
345           for (String roleStr: roleList) {
346             roles.add(new JaxbRole(roleStr, jaxbOrganization, "Brightspace external role", Role.Type.EXTERNAL));
347             if (roleStr.endsWith(LTI_INSTRUCTOR_ROLE)) {
348               isInstructor = true;
349             }
350           }
351           roles.add(new JaxbRole(Group.ROLE_PREFIX + "BRIGHTSPACE", jaxbOrganization, "Brightspace User",
352                   Role.Type.EXTERNAL_GROUP));
353           if (isInstructor) {
354             roles.add(new JaxbRole(Group.ROLE_PREFIX + "BRIGHTSPACE_INSTRUCTOR", jaxbOrganization,
355                       "Brightspace Instructor", Role.Type.EXTERNAL_GROUP));
356           }
357           logger.debug("Returning JaxbRoles: {}", roles);
358 
359           User user =  new JaxbUser(username, null, brightspaceUser.getDisplayName(),
360                   brightspaceUser.getExternalEmail(), this.getName(), jaxbOrganization, roles);
361           cache.put(username, user);
362           logger.debug("Returning user {}", user);
363           return user;
364         } else {
365           cache.put(username, nullToken);
366           logger.debug("User {} not found in Brightspace system", username);
367           return null;
368         }
369       } catch (BrightspaceClientException e) {
370         logger.error("A Brightspace API error ( {} ) occurred, user {} could not be retrieved", e, username);
371         return null;
372       } finally {
373         currentThread.setContextClassLoader(originalClassloader);
374       }
375     }
