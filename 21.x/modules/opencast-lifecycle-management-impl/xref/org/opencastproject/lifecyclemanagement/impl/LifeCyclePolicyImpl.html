61  /**
62   * Entity object for storing life cycle policies in persistence storage. Policies define actions that should be
63   * applied to entities. They offer various methods of defining when an action should be applied, effectively
64   * allowing for scheduling of actions.
65   *
66   * The policy ACL only pertains to itself. The entities they may affect (e.g. events, series) are still only governed
67   * by their own ACLs.
68   */
69  @Entity(name = "LifeCyclePolicy")
70  @Table(name = "oc_lifecyclepolicy")
71  @NamedQueries({
72      @NamedQuery(
73          name = "LifeCyclePolicy.findById",
74          query = "SELECT p FROM LifeCyclePolicy p WHERE p.id = :id and p.organization = :organizationId"
75      ),
76      @NamedQuery(
77          name = "LifeCyclePolicy.findActive",
78          query = "SELECT p FROM LifeCyclePolicy p WHERE p.isActive = true and p.organization = :organizationId"
79      ),
80      @NamedQuery(
81          name = "LifeCyclePolicy.deleteAllCreatedFromConfig",
82          query = "DELETE FROM LifeCyclePolicy p WHERE p.isCreatedFromConfig = true "
83              + "and p.organization = :organizationId"
84      ),
85      @NamedQuery(
86          name = "LifeCyclePolicy.getTotal",
87          query = "SELECT COUNT(p) FROM LifeCyclePolicy p WHERE p.organization = :organizationId"
88      ),
89  })
90  public class LifeCyclePolicyImpl implements LifeCyclePolicy {
91  
92    @Id
93    @GeneratedValue(strategy = GenerationType.AUTO)
94    @Column(name = "id")
95    private String id;
96  
97    @Column(name = "organization", nullable = false, length = 128)
98    private String organization;
99  
100   @Column(name = "title")
101   private String title;
102 
103   @Column(name = "target_type")
104   private TargetType targetType;
105 
106   @Column(name = "action")
107   private Action action;
108 
109   // JSON
110   @Lob
111   @Column(name = "action_parameters")
112   private String actionParameters;
113 
114   @Column(name = "actionDate")
115   @Temporal(TemporalType.TIMESTAMP)
116   private Date actionDate;
117 
118   @Column(name = "cronTrigger")
119   private String cronTrigger;
120 
121   @Column(name = "timing")
122   private Timing timing;
123 
124   @Column(name = "isActive")
125   private boolean isActive = true;
126 
127   @Column(name = "isCreatedFromConfig")
128   private boolean isCreatedFromConfig = false;
129 
130   // JSON
131   @Lob
132   @Column(name = "targetFilters")
133   private String targetFilters;
134 
135   @OneToMany(
136       fetch = FetchType.LAZY,
137       cascade = { CascadeType.PERSIST, CascadeType.MERGE, CascadeType.REMOVE },
138       orphanRemoval = true
139   )
140   @JoinColumn(name = "lifecyclepolicy_id", nullable = false)
141   @OrderColumn(name = "position_policy_access_control_entries")
142   @CascadeOnDelete
143   private List<LifeCyclePolicyAccessControlEntryImpl> accessControlEntries = new ArrayList<>();
144 
145   @Transient
146   private static final Gson gson = new Gson();
147 
148   public LifeCyclePolicyImpl() {
149 
150   }

152   public LifeCyclePolicyImpl(String title, TargetType targetType, Action action, String actionParameters,
153       Date actionDate, String cronTrigger, Timing timing,
154       Map<String, Map<String, EventSearchQueryField<String>>>  targetFilters,
155       List<LifeCyclePolicyAccessControlEntryImpl> accessControlEntries) {
156     this.title = title;
157     this.targetType = targetType;
158     this.action = action;
159     this.actionParameters = actionParameters;
160     this.actionDate = actionDate;
161     this.cronTrigger = cronTrigger;
162     this.timing = timing;
163     setTargetFilters(targetFilters);
164     this.accessControlEntries = accessControlEntries;
165   }
